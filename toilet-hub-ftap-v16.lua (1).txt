local function HsvToRgb(h, s, v)
    local c = v * s
    local x = c * (1 - math.abs(math.fmod(h / 60, 2) - 1))
    local m = v - c
    local r, g, b
    if h >= 0 and h < 60 then r, g, b = c, x, 0
    elseif h >= 60 and h < 120 then r, g, b = x, c, 0
    elseif h >= 120 and h < 180 then r, g, b = 0, c, x
    elseif h >= 180 and h < 240 then r, g, b = 0, x, c
    elseif h >= 240 and h < 300 then r, g, b = x, 0, c
    elseif h >= 300 and h < 360 then r, g, b = c, 0, x
    else r, g, b = 0, 0, 0 end
    return Color3.new(r + m, g + m, b + m)
end

local function getNearestPlayer()
    local nearestPlayer, shortestDistance = nil, math.huge
    for _, player in pairs(Services.Players:GetPlayers()) do
        if player ~= PlayerVars.plr and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local distance = (PlayerVars.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
            if distance < shortestDistance then
                nearestPlayer, shortestDistance = player, distance
            end
        end
    end
    return nearestPlayer
end

local function getPlayersInRange(center, radius)

            local playerTorso = player.Character:FindFirstChild("Torso") or player.Character:FindFirstChild("HumanoidRootPart")
            if playerTorso then

local function fireLineToPlayer(part, repeatCount)
    local players = Services.Players:GetPlayers()
    if #players == 0 or not part then return end
    local playerIndex = 1
    for _ = 1, math.min(repeatCount, 1000000) do
        if playerIndex > #players then playerIndex = 1 end
        local player = players[playerIndex]
        if player and player.Character and player.Character:FindFirstChild("Head") then
            local targetPos = player.Character.Head.Position
            pcall(function() ServerEvents.CreateLine:FireServer(part, CFrame.new(targetPos)) end)
        end
        playerIndex = playerIndex + 1
    end
end

local function startTeleportLoop()
    if Vars.loopThread then return end
    Vars.teleportConfig.loop = true
    Vars.loopThread = task.spawn(function()
        while Vars.teleportConfig.loop do
            local players = Services.Players:GetPlayers()
            if Vars.teleportConfig.randomOrder then
                for i = #players, 2, -1 do
                    local j = math.random(1, i)
                    players[i], players[j] = players[j], players[i]
                end
            end
            for _, pl in ipairs(players) do
                if not Vars.teleportConfig.loop then break end
                if pl ~= PlayerVars.plr then
                    local tgtHRP = getHRP(pl.Character)
                    local myHRP = getHRP(PlayerVars.plr.Character)
                    if tgtHRP and myHRP then
                        local dest = tgtHRP.CFrame + Vector3.new(0, Vars.teleportConfig.heightOffset, 0)
                        pcall(function() myHRP.CFrame = dest end)
                        task.wait(math.clamp(Vars.teleportConfig.delay, Vars.teleportConfig.minDelay, Vars.teleportConfig.maxDelay))
                    elseif not tgtHRP and not Vars.teleportConfig.skipIfNoHRP then
                        -- スキップ
                    end
                end
            end
            task.wait(0)
        end
        Vars.loopThread = nil
    end)
end

local function stopTeleportLoop()
    Vars.teleportConfig.loop = false
    Vars.loopThread = nil
end

-- Anti Blobman関数
local function getBlobmanOwner(blobman)
    if not blobman then return nil end
    for _, player in pairs(Services.Players:GetPlayers()) do
        local toyFolder = Services.Workspace:FindFirstChild(player.Name .. "SpawnedInToys")
        if toyFolder and blobman:IsDescendantOf(toyFolder) then
            return player
        end
    end
    return nil
end

local function findGrabbingBlobman()
    local myRoot = getRoot()
    if not myRoot then return nil end
    for _, toyFolder in pairs(Services.Workspace:GetChildren()) do
        if toyFolder.Name:match("SpawnedInToys$") then
            for _, toy in pairs(toyFolder:GetChildren()) do
                if toy.Name == "CreatureBlobman" then
                    local leftDetector = toy:FindFirstChild("LeftDetector")
                    local rightDetector = toy:FindFirstChild("RightDetector")
                    if leftDetector then
                        local leftWeld = leftDetector:FindFirstChild("LeftWeld")
                        if leftWeld and leftWeld.Part1 and leftWeld.Part1:IsDescendantOf(PlayerVars.plr.Character) then
                            return toy
                        end
                    end
                    if rightDetector then
                        local rightWeld = rightDetector:FindFirstChild("RightWeld")
                        if rightWeld and rightWeld.Part1 and rightWeld.Part1:IsDescendantOf(PlayerVars.plr.Character) then
                            return toy
                        end
                    end
                end
            end
        end
    end
    return nil
end

        if part:IsA("BasePart") then

            bv.Velocity = Vector3.new(math.random(-1,1) * 1e6, 1e6, math.random(-1,1) * 1e6)
            bv.MaxForce = Vector3.one * math.huge
            Services.Debris:AddItem(bv, 3)
        end
    end
end

local function getAllBlobmanOperators()
    local operators = {}
    for _, player in pairs(Services.Players:GetPlayers()) do
        if player ~= PlayerVars.plr then
            local toyFolder = Services.Workspace:FindFirstChild(player.Name .. "SpawnedInToys")
            if toyFolder then
                for _, toy in pairs(toyFolder:GetChildren()) do
                    if toy.Name == "CreatureBlobman" then
                        local seat = toy:FindFirstChild("VehicleSeat")
                        if seat and seat:FindFirstChild("SeatWeld") then
                            operators[player] = true
                            break
                        end
                    end
                end
            end
        end
    end
    return operators
end

local function kickAllBlobmanOperators()
    local operators = getAllBlobmanOperators()
    for player, _ in pairs(operators) do
        if player and player.Character then
            AttackFunctions.killPlayer(player)
            AttackFunctions.flingPlayer(player)
            AttackFunctions.voidPlayer(player)
        end
    end
end

    local hrp = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    pcall(function()
        if Vars.blobalter == 1 then
            blobman.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(blobman.LeftDetector, hrp, blobman.LeftDetector.LeftWeld)
            Vars.blobalter = 2
        else
            blobman.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(blobman.RightDetector, hrp, blobman.RightDetector.RightWeld)
            Vars.blobalter = 1
        end
    end)
end

local function getBlobman()
    local inv = getInv()
    if inv then
        local b = inv:FindFirstChild(Vars.BLOB_NAME)
        if b and b:FindFirstChild(Vars.SEAT_NAME) then return b end
    end
    return nil
end

end

-- Kick Aura関数
local function destroyKickAuraBlobman()
    if Vars.kickAuraBlobman and Vars.kickAuraBlobman.Parent then
        pcall(function() ServerEvents.DestroyToy:FireServer(Vars.kickAuraBlobman) end)
    end
    Vars.kickAuraBlobman = nil
end

local function kickAuraBlobGrab(blob, target, side)
    local detector = blob:FindFirstChild(side .. "Detector")
    local weld = detector and detector:FindFirstChild(side .. "Weld")
    if detector and weld then blob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(detector, target, weld) end
end

local function kickAuraUngrab(part) ServerEvents.DestroyLine:FireServer(part) end

local function kickAuraBlobKick(blob, target, side)
    kickAuraBlobGrab(blob, getRoot(), side)
    task.wait(0.1)
    SetNetworkOwnerFunc(target)
    task.wait()
    target.CFrame = target.CFrame + Vector3.new(0, 10, 0)
    kickAuraUngrab(target)
    task.wait(0.1)
    kickAuraBlobGrab(blob, target, side)
end

local function removeAllForcesKickAura(character)
    for _, v in ipairs(character:GetDescendants()) do
        if v:IsA("BodyForce") or v:IsA("BodyVelocity") or v:IsA("BodyAngularVelocity") or v:IsA("BodyGyro") or v:IsA("VectorForce") or v:IsA("LinearVelocity") or v:IsA("AngularVelocity") or v:IsA("AlignPosition") or v:IsA("AlignOrientation") then
            v:Destroy()
        end
    end
end

local function ensureKickAuraBlobmanAndSit()
    local hum = getHum()
    local root = getRoot()
    if not (hum and root) then return end
    if hum.Sit then return end
    if not (Vars.kickAuraBlobman and Vars.kickAuraBlobman.Parent) then
        pcall(function() Services.ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer("CreatureBlobman", root.CFrame * CFrame.new(0, 0, -5), Vector3.new(0, -15.716, 0)) end)
        task.wait(0.2)
        local inv = getInv()
        if inv then
            for _, v in ipairs(inv:GetChildren()) do
                if v.Name == "CreatureBlobman" and v:FindFirstChildOfClass("VehicleSeat") then Vars.kickAuraBlobman = v; break end
            end
        end
    end
    local seat = Vars.kickAuraBlobman and Vars.kickAuraBlobman:FindFirstChildOfClass("VehicleSeat")
    if seat and not hum.Sit then seat:Sit(hum) end
end

local function shouldProcessPlayerKickAura(player)
    if player == PlayerVars.plr then return false end
    if Vars.kickFriendWhitelist and isFriendCheck(player) then return false end
    local inPlot = player:FindFirstChild("InPlot")
    if inPlot and inPlot.Value then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local last = Vars.KickCooldown[player.UserId]
    if last and (os.clock() - last) < Vars.KICK_COOLDOWN_TIME then return false end
    return true
end

local function startKickAuraLoop()
    if Vars.kickAuraLoop then Vars.kickAuraLoop:Disconnect() end
    Vars.kickAuraLoop = Services.RunService.Heartbeat:Connect(function()
        if not Vars.kickAuraEnabled then return end
        ensureKickAuraBlobmanAndSit()
        if not Vars.kickAuraBlobman then return end
        local root = getRoot()
        if not root then return end
        if Vars.kickAuraTargetMode and Vars.TargetPlayer then
            if shouldProcessPlayerKickAura(Vars.TargetPlayer) then
                local char = Vars.TargetPlayer.Character
                local targetRoot = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
                if targetRoot then
                    kickAuraBlobKick(Vars.kickAuraBlobman, targetRoot, "Left")
                    Vars.ForceDisabledPlayers[Vars.TargetPlayer] = true
                    removeAllForcesKickAura(char)
                    Vars.KickCooldown[Vars.TargetPlayer.UserId] = os.clock()
                end
            end
        else
            for _, player in ipairs(Services.Players:GetPlayers()) do
                if shouldProcessPlayerKickAura(player) then
                    local char = player.Character
                    local targetRoot = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
                    if targetRoot and (targetRoot.Position - root.Position).Magnitude <= Vars.kickAuraRange then
                        kickAuraBlobKick(Vars.kickAuraBlobman, targetRoot, "Left")
                        Vars.ForceDisabledPlayers[player] = true
                        removeAllForcesKickAura(char)
                        Vars.KickCooldown[player.UserId] = os.clock()
                        break
                    end
                end
            end
        end
    end)
end

local function cleanupKickAura()
    if Vars.kickAuraLoop then Vars.kickAuraLoop:Disconnect(); Vars.kickAuraLoop = nil end
    Vars.KickCooldown = {}
    Vars.ForceDisabledPlayers = {}
    destroyKickAuraBlobman()
end

-- Wing System関数
local function CleanupWings()
    for _, w in pairs(Vars.Wings) do
        if w.Handle then w.Handle:Destroy() end
        for _, s in pairs(w.Segments) do if s.Part then s.Part:Destroy() end end
    end
    Vars.Wings = {}
end

local function GetToyFolders()
    local folders = {}
    if Vars.WingSearchMode == "My Toys" or Vars.WingSearchMode == "All Toys" then
        local myFolder = Services.Workspace:FindFirstChild(PlayerVars.plr.Name.."SpawnedInToys")
        if myFolder then table.insert(folders, myFolder) end
    end
    if Vars.WingSearchMode == "Plot Toys" or Vars.WingSearchMode == "All Toys" then
        local plotsFolder = Services.Workspace:FindFirstChild("Plots")
        if plotsFolder then
            for i=1, 5 do
                local plot = plotsFolder:FindFirstChild("Plot"..i)
                if plot then
                    local ownersFolder = plot:FindFirstChild("PlotSign") and plot.PlotSign:FindFirstChild("ThisPlotsOwners")
                    if ownersFolder then
                        for _, v in ipairs(ownersFolder:GetChildren()) do
                            if v:IsA("ValueBase") and v.Value == PlayerVars.plr.Name then
                                local plotItemsFolder = Services.Workspace:FindFirstChild("PlotItems")
                                if plotItemsFolder and plotItemsFolder:FindFirstChild(plot.Name) then
                                    table.insert(folders, plotItemsFolder:FindFirstChild(plot.Name))
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return folders
end

local function SetupWingPhysics(Part)
    local BP = Part:FindFirstChildOfClass("BodyPosition") or Instance.new("BodyPosition")
    local BG = Part:FindFirstChildOfClass("BodyGyro") or Instance.new("BodyGyro")
    BP.P = 15000; BP.D = 200; BP.MaxForce = Vector3.one * 1e10; BP.Parent = Part
    BG.P = 15000; BG.D = 200; BG.MaxTorque = Vector3.one * 1e10; BG.Parent = Part
    return BG, BP
end

local function BuildWings()
    CleanupWings()
    local folders = GetToyFolders()
    local allSpawnedToys = {}
    for _, folder in ipairs(folders) do
        for _, x in ipairs(folder:GetDescendants()) do
            if x:IsA("Model") and x.Name == Vars.SelectedWingItem then
                table.insert(allSpawnedToys, x)
            end
        end
    end
    if #allSpawnedToys == 0 then return false end
    for i = 1, 2 do
        local Segments = {}
        for x = 1, Vars.WingLength do
            local p = Instance.new("Part")
            p.CanCollide = false; p.Anchored = true; p.Transparency = 1; p.Size = Vector3.new(4,1,4); p.Parent = Services.Workspace
            Segments[#Segments+1] = {Part = p}
        end
        local h = Instance.new("Part")
        h.CanCollide = false; h.Anchored = true; h.Transparency = 1; h.Size = Vector3.new(4,1,4); h.Parent = Services.Workspace
        Vars.Wings[#Vars.Wings+1] = {Handle = h, Segments = Segments, Sync = {}, Reserved = nil}
    end
    for i, v in ipairs(allSpawnedToys) do
        local Side = (i <= (#allSpawnedToys/2)) and 1 or 2
        local Pallet = v:FindFirstChild("SoundPart") or v:FindFirstChild("Handle") or v:FindFirstChildWhichIsA("BasePart")
        if Pallet then
            for _, child in pairs(v:GetChildren()) do if child:IsA("BasePart") then child.CanCollide = false end end
            local BG, BP = SetupWingPhysics(Pallet)
            if not Vars.Wings[Side].Reserved then Vars.Wings[Side].Reserved = {BG = BG, BP = BP}
            else table.insert(Vars.Wings[Side].Sync, {BG = BG, BP = BP}) end
        end
    end
    return true
end

-- Grab関数
local function handleKillGrab(model)
    local part_to_impulse = model:FindFirstChild("GrabPart") and model.GrabPart:FindFirstChild("WeldConstraint") and model.GrabPart.WeldConstraint.Part1
    if part_to_impulse then part_to_impulse.Parent:BreakJoints() end  
end

local function handleSkyGrab(model)
    local part_to_impulse = model:FindFirstChild("GrabPart")
    if part_to_impulse and part_to_impulse:FindFirstChild("WeldConstraint") then
        local bodyVelocity = Instance.new("BodyVelocity")
        bodyVelocity.Parent = part_to_impulse
        bodyVelocity.Velocity = Vector3.new(0, 100, 0)
        bodyVelocity.MaxForce = Vector3.new(0, math.huge, 0)
    end
end

local function handleSpinGrab(model)
    local part_to_impulse = model:FindFirstChild("GrabPart") and model.GrabPart:FindFirstChild("WeldConstraint") and model.GrabPart.WeldConstraint.Part1
    if part_to_impulse then
        local bodyAngularVelocity = Instance.new("BodyAngularVelocity")
        bodyAngularVelocity.Parent = part_to_impulse
        bodyAngularVelocity.AngularVelocity = Vector3.new(0, 1000, 0)
        bodyAngularVelocity.MaxTorque = Vector3.one * math.huge
    end
end

local function handleFreezeGrab(model)
    local part_to_impulse = model:FindFirstChild("GrabPart")
    if part_to_impulse then
        local bodyPosition = Instance.new("BodyPosition")
        bodyPosition.Parent = part_to_impulse
        bodyPosition.Position = part_to_impulse.Position
        bodyPosition.MaxForce = Vector3.one * math.huge
    end
end

-- Toilet関数
local function createToilet(position)
    local toilet = Instance.new("Model")
    toilet.Name = "ToiletHub_Toilet"
    local base = Instance.new("Part")
    base.Name = "ToiletBase"; base.Size = Vector3.new(3, 2, 4); base.Color = Color3.fromRGB(255, 255, 255)
    base.Material = Enum.Material.SmoothPlastic; base.Anchored = true; base.Position = position; base.Parent = toilet
    local water = Instance.new("Part")
    water.Name = "ToiletWater"; water.Size = Vector3.new(2, 0.5, 2); water.Color = Color3.fromRGB(100, 150, 255)
    water.Material = Enum.Material.Glass; water.Transparency = 0.3; water.Anchored = true
    water.Position = position + Vector3.new(0, 0.8, 0); water.Parent = toilet
    toilet.Parent = Services.Workspace
    return toilet
end

local function flushToilet(toilet, power)
    if not toilet then return end
    local water = toilet:FindFirstChild("ToiletWater")
    if water then
        local spinSpeed = power == "Large" and 50 or 25
        local bav = Instance.new("BodyAngularVelocity")
        bav.AngularVelocity = Vector3.new(0, spinSpeed, 0)
        bav.MaxTorque = Vector3.one * math.huge
        bav.Parent = water
        Services.Debris:AddItem(bav, 3)
    end
end

Services.RunService.RenderStepped:Connect(function(dt)
    Vars.RainbowTimer = math.fmod(Vars.RainbowTimer + dt * 100, 360)
    local RainbowColor = HsvToRgb(Vars.RainbowTimer, 1, 1)
    local myRoot = getRoot()

    -- Speed Boost
    if asss and PlayerVars.plr.Character and PlayerVars.plr.Character:FindFirstChild("HumanoidRootPart") and PlayerVars.plr.Character:FindFirstChild("Humanoid") then
        PlayerVars.plr.Character.HumanoidRootPart.CFrame = PlayerVars.plr.Character.HumanoidRootPart.CFrame + PlayerVars.plr.Character.Humanoid.MoveDirection * (wss / 10)
    end

    -- FOV Circle
    if Vars.FOVCircle then
        pcall(function()
            Vars.FOVCircle.Position = Services.UserInputService:GetMouseLocation()
            if Vars._G_FOVRainbow then Vars.FOVCircle.Color = RainbowColor else Vars.FOVCircle.Color = Vars._G_FOVColor end
        end)
    end

    -- PC Silent Aim
    if Vars._G_SilentEnabled and Vars.CameraClone and Vars.Camera then
        pcall(function()
            local TargetCFrame = Vars.CameraClone.CFrame
            Vars.TargetHRP = nil
            if not Services.Workspace:FindFirstChild("GrabParts") then
                local Position = Vars.CameraClone.CFrame.Position + (Vars.CameraClone.CFrame.LookVector * 10)
                local Distance = Vars._G_ReachOverride
                local Target = nil
                for HRPart, _ in pairs(Vars.HRPs) do
                    local Player = GetPlayerFromHRP(HRPart)
                    if Player and Player ~= PlayerVars.plr and HRPart.Parent and HRPart.Parent:FindFirstChild("Humanoid") and HRPart.Parent.Humanoid.Health > 0 then
                        local Magnitude = (HRPart.Position - Position).Magnitude
                        if Magnitude < Distance then Distance = Magnitude; Target = HRPart end
                    end
                end
                if Target and Target ~= myRoot then
                    Vars.TargetHRP = Target
                    if TargetCFrame.LookVector:Dot((Target.Position - TargetCFrame.Position).Unit) > 0 then
                        local Radius = Vars._G_FOVOverride
                        local TargetHead = Target.Parent:FindFirstChild("Head") or Target
                        local Screen3D = Vars.CameraClone:WorldToScreenPoint(TargetHead.Position)
                        local Screen2D = Vector2.new(Screen3D.X, Screen3D.Y)
                        local Center = Vars.Camera.ViewportSize / 2
                        if (Screen2D - Center).Magnitude <= Radius then
                            TargetCFrame = CFrame.new(TargetCFrame.Position, TargetHead.Position)
                        else Vars.TargetHRP = nil end
                    else Vars.TargetHRP = nil end
                end
            end
            Vars.Camera.CFrame = TargetCFrame
        end)
    elseif Vars.CameraClone and Vars.Camera then
        pcall(function() Vars.Camera.CFrame = Vars.CameraClone.CFrame end)
    end

    -- Wing Animation
    if Vars.WingsActive and #Vars.Wings > 0 then
        local Char = PlayerVars.plr.Character
        local Torso = Char and (Char:FindFirstChild("Torso") or Char:FindFirstChild("UpperTorso"))
        local HumanoidRoot = Char and Char:FindFirstChild("HumanoidRootPart")
        if Torso and HumanoidRoot then
            Vars.TimeCounter = Vars.TimeCounter + dt * (Vars.WingSpeed + HumanoidRoot.Velocity.Magnitude / 40)
            for i, Wing in ipairs(Vars.Wings) do
                local direction = (i == 1) and 1 or -1
                local flap = math.sin(Vars.TimeCounter) * math.rad(Vars.WingAngle + (HumanoidRoot.Velocity.Magnitude/4)) * direction
                Wing.Handle.CFrame = Torso.CFrame * Vars.WingOffsets[i] * CFrame.Angles(0, 0, flap)
                if Wing.Reserved then
                    Wing.Reserved.BP.Position = Wing.Handle.Position
                    Wing.Reserved.BG.CFrame = Wing.Handle.CFrame * CFrame.Angles(math.rad(90), 0, math.rad(90))
                end
                for Index, Segment in ipairs(Wing.Segments) do
                    local ToFollow = (Index == 1) and Wing.Handle.CFrame or Wing.Segments[Index-1].Part.CFrame
                    Segment.Part.CFrame = Segment.Part.CFrame:Lerp(ToFollow * Vars.WingOffsets[i], 0.5)
                    if Wing.Sync[Index] then
                        Wing.Sync[Index].BP.Position = Segment.Part.Position
                        Wing.Sync[Index].BG.CFrame = Segment.Part.CFrame * CFrame.Angles(math.rad(90), 0, math.rad(90))
                    end
                end
            end
        end
    end

    -- ESP Updates
    local LineColor = Vars._G_LineRainbow and RainbowColor or Vars._G_LineColor
    local NameColor = Vars._G_NameRainbow and RainbowColor or Vars._G_NameColor
    local BoxColor = Vars._G_BoxRainbow and RainbowColor or Vars._G_BoxColor

    for Player, esp in pairs(Vars.ESPDrawings) do
        pcall(function()
            local Char = Player.Character
            local HRPart = Char and Char:FindFirstChild("HumanoidRootPart")
            local Head = Char and Char:FindFirstChild("Head")
            local Hum = Char and Char:FindFirstChild("Humanoid")
            local IsDead = not Hum or Hum.Health <= 0
            local shouldDraw = HRPart and not IsDead and Player ~= PlayerVars.plr and Head and myRoot
            
            if not shouldDraw then
                if esp.Line then esp.Line.Visible = false end
                if esp.Name then esp.Name.Visible = false end
                if esp.Box then esp.Box.Visible = false end
                if esp.HealthBar then esp.HealthBar.Visible = false end
                if esp.HealthBarBG then esp.HealthBarBG.Visible = false end
            else
                local ScreenPosition, OnScreen = Vars.Camera:WorldToScreenPoint(Head.Position)
                if Vars._G_LineEnabled and esp.Line then
                    if OnScreen then esp.Line.From = Services.UserInputService:GetMouseLocation(); esp.Line.To = Vector2.new(ScreenPosition.X, ScreenPosition.Y); esp.Line.Color = LineColor; esp.Line.Visible = true
                    else esp.Line.Visible = false end
                elseif esp.Line then esp.Line.Visible = false end
                if Vars._G_NameEnabled and esp.Name then
                    if OnScreen then
                        local Dist = math.floor((HRPart.Position - myRoot.Position).Magnitude)
                        esp.Name.Position = Vector2.new(ScreenPosition.X, ScreenPosition.Y - 20)
                        esp.Name.Text = string.format("%s [%d]", Player.DisplayName, Dist)
                        esp.Name.Color = NameColor; esp.Name.Visible = true
                    else esp.Name.Visible = false end
                elseif esp.Name then esp.Name.Visible = false end
                if Vars._G_BoxEnabled and esp.Box then
                    if OnScreen then
                        local Size = 1000 / ScreenPosition.Z
                        esp.Box.Position = Vector2.new(ScreenPosition.X - Size/2, ScreenPosition.Y - Size)
                        esp.Box.Size = Vector2.new(Size, Size * 2); esp.Box.Color = BoxColor; esp.Box.Visible = true
                    else esp.Box.Visible = false end
                elseif esp.Box then esp.Box.Visible = false end
                if Vars._G_HealthBarEnabled and esp.HealthBar and esp.HealthBarBG and Hum then
                    if OnScreen then
                        local Size = 1000 / ScreenPosition.Z
                        local HealthPercent = Hum.Health / Hum.MaxHealth
                        local BarHeight = Size * 2
                        local BarX = ScreenPosition.X - Size/2 - 5
                        local BarY = ScreenPosition.Y - Size
                        esp.HealthBarBG.From = Vector2.new(BarX, BarY); esp.HealthBarBG.To = Vector2.new(BarX, BarY + BarHeight); esp.HealthBarBG.Visible = true
                        esp.HealthBar.From = Vector2.new(BarX, BarY + BarHeight * (1 - HealthPercent)); esp.HealthBar.To = Vector2.new(BarX, BarY + BarHeight)
                        esp.HealthBar.Color = Color3.fromRGB(255 * (1 - HealthPercent), 255 * HealthPercent, 0); esp.HealthBar.Visible = true
                    else esp.HealthBar.Visible = false; if esp.HealthBarBG then esp.HealthBarBG.Visible = false end end
                elseif esp.HealthBar then esp.HealthBar.Visible = false; if esp.HealthBarBG then esp.HealthBarBG.Visible = false end end
            end
        end)
    end

    -- Kick Aura Force Removal
    if Vars.kickAuraEnabled then
        for player in pairs(Vars.ForceDisabledPlayers) do
            local char = player.Character
            if char then removeAllForcesKickAura(char) end
        end
    end
end)

                local targetPart = MobileTarget.Character:FindFirstChild(partName)
                if distance <= Vars._G_MobileSilentDist and targetPart then
                    args[3] = CalculateDirection(args[2], targetPart.Position, 1000)
                    args[4] = RaycastParams.new(); args[4].FilterDescendantsInstances = {MobileTarget.Character}; args[4].FilterType = Enum.RaycastFilterType.Include
                end
            end
            return __namecall(unpack(args))
        end)
    end
end)
