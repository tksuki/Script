local function AddConnection(Signal, Function)
	if (not OrionLib:IsRunning()) then
		return
	end
	local SignalConnect = Signal:Connect(Function)
	table.insert(OrionLib.Connections, SignalConnect)
	return SignalConnect
end

		task.wait()
	end

	end
end)

local function MakeDraggable(DragPoint, Main, FullScreenRef, MinimizedRef)
	pcall(function()
		local Dragging, DragInput, MousePos, FramePos = false
		AddConnection(DragPoint.InputBegan, function(Input)
			if (FullScreenRef and FullScreenRef()) then
				return
			end
			if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
				Dragging = true
				MousePos = Input.Position
				FramePos = Main.Position
				Input.Changed:Connect(function()
					if Input.UserInputState == Enum.UserInputState.End then
						Dragging = false
					end
				end)
			end
		end)
		AddConnection(DragPoint.InputChanged, function(Input)
			if (FullScreenRef and FullScreenRef()) then
				return
			end
			if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch then
				DragInput = Input
			end
		end)
		AddConnection(UserInputService.InputChanged, function(Input)
			if (FullScreenRef and FullScreenRef()) then
				return
			end
			if Input == DragInput and Dragging then
				local Delta = Input.Position - MousePos
				Main.Position  = UDim2.new(FramePos.X.Scale, FramePos.X.Offset + Delta.X, FramePos.Y.Scale, FramePos.Y.Offset + Delta.Y)
			end
		end)
	end)
end

local function MakeResizable(ResizePoint, Main, MinSize, MaxSize, FullScreenRef, MinimizedRef, ResizeSide)
    pcall(function()
        local Resizing, ResizeInput, MousePos, FrameSize = false
        MinSize = MinSize or UDim2.new(0, 450, 0, 350)
        MaxSize = MaxSize or UDim2.new(0, 1400, 0, 900)
        ResizeSide = ResizeSide or "both"

        AddConnection(ResizePoint.InputBegan, function(Input)
            if (FullScreenRef and FullScreenRef()) or (MinimizedRef and MinimizedRef()) then
                return
            end
            if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
                Resizing = true
                MousePos = Input.Position
                FrameSize = Main.Size
                Input.Changed:Connect(function()
                    if Input.UserInputState == Enum.UserInputState.End then
                        Resizing = false
                    end
                end)
            end
        end)

        AddConnection(ResizePoint.InputChanged, function(Input)
            if (FullScreenRef and FullScreenRef()) or (MinimizedRef and MinimizedRef()) then
                return
            end
            if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch then
                ResizeInput = Input
            end
        end)

        AddConnection(UserInputService.InputChanged, function(Input)
            if Input == ResizeInput and Resizing then
                local Delta = Input.Position - MousePos
                local NewWidth = FrameSize.X.Offset
                local NewHeight = FrameSize.Y.Offset

                if ResizeSide == "both" or ResizeSide == "horizontal" then
                    NewWidth = math.clamp(FrameSize.X.Offset + Delta.X, MinSize.X.Offset, MaxSize.X.Offset)
                end
                if ResizeSide == "both" or ResizeSide == "vertical" then
                    NewHeight = math.clamp(FrameSize.Y.Offset + Delta.Y, MinSize.Y.Offset, MaxSize.Y.Offset)
                end

                Main.Size = UDim2.new(0, NewWidth, 0, NewHeight)
            end
        end)
    end)
end

local function Create(Name, Properties, Children)
	local Object = Instance.new(Name)
	for i, v in next, Properties or {} do
		Object[i] = v
	end
	for i, v in next, Children or {} do
		v.Parent = Object
	end
	return Object
end

local function SetProps(Element, Props)
	table.foreach(Props, function(Property, Value)
		Element[Property] = Value
	end)
	return Element
end

local function SetChildren(Element, Children)
	table.foreach(Children, function(_, Child)
		Child.Parent = Element
	end)
	return Element
end

local function Round(Number, Factor)
	local Result = math.floor(Number/Factor + (math.sign(Number) * 0.5)) * Factor
	if Result < 0 then Result = Result + Factor end
	return Result
end

local function ReturnProperty(Object)
	if Object:IsA("Frame") or Object:IsA("TextButton") then
		return "BackgroundColor3"
	end
	if Object:IsA("ScrollingFrame") then
		return "ScrollBarImageColor3"
	end
	if Object:IsA("UIStroke") then
		return "Color"
	end
	if Object:IsA("TextLabel") or Object:IsA("TextBox") then
		return "TextColor3"
	end
	if Object:IsA("ImageLabel") or Object:IsA("ImageButton") then
		return "ImageColor3"
	end
end

local function AddThemeObject(Object, Type)
	if not OrionLib.ThemeObjects[Type] then
		OrionLib.ThemeObjects[Type] = {}
	end
	table.insert(OrionLib.ThemeObjects[Type], Object)
	Object[ReturnProperty(Object)] = OrionLib.Themes[OrionLib.SelectedTheme][Type]
	return Object
end

local function SetTheme()
	for Name, Type in pairs(OrionLib.ThemeObjects) do
		for _, Object in pairs(Type) do
			Object[ReturnProperty(Object)] = OrionLib.Themes[OrionLib.SelectedTheme][Name]
		end
	end
end

local function PackColor(Color)
	return {R = Color.R * 255, G = Color.G * 255, B = Color.B * 255}
end

local function UnpackColor(Color)
	return Color3.fromRGB(Color.R, Color.G, Color.B)
end

local function LoadCfg(Config)
	local Data = HttpService:JSONDecode(Config)
	table.foreach(Data, function(a,b)
		if OrionLib.Flags[a] then
			task.spawn(function()
				if OrionLib.Flags[a].Type == "Colorpicker" then
					OrionLib.Flags[a]:Set(UnpackColor(b))
				else
					OrionLib.Flags[a]:Set(b)
				end
			end)
		end
	end)
end

local function SaveCfg(Name)
	local Data = {}
	for i,v in pairs(OrionLib.Flags) do
		if v.Save then
			if v.Type == "Colorpicker" then
				Data[i] = PackColor(v.Value)
			else
				Data[i] = v.Value
			end
		end
	end
	writefile(OrionLib.Folder .. "/" .. Name .. ".txt", tostring(HttpService:JSONEncode(Data)))
end

local function UnlockMouse(Value)
	if Value then
		mouselock = true
		task.spawn(function()
			while mouselock do
				UserInputService.MouseIconEnabled = Value
				freeMouse.Visible = Value
				task.wait()
			end
			UserInputService.MouseIconEnabled = false
			freeMouse.Visible = false
		end)
	else
		mouselock = false
	end
end

		pcall(function()
			if isfile(OrionLib.Folder .. "/" .. game.GameId .. ".txt") then
				LoadCfg(readfile(OrionLib.Folder .. "/" .. game.GameId .. ".txt"))
				OrionLib:MakeNotification({
					Name = "Configuration Loaded",
					Content = "Successfully loaded configuration for game " .. game.GameId .. ".",
					Time = 5
				})
			end
		end)
	end
end

	local function SearchElements(SearchText)
		SearchText = string.lower(SearchText)
		if SearchText == "" then
			for _, Element in pairs(AllElements) do
				if Element.Frame then
					Element.Frame.Visible = true
				end
			end
			for _, Tab in pairs(TabHolder:GetChildren()) do
				if Tab:IsA("TextButton") then
					Tab.Visible = true
				end
			end
			return
		end
		for _, Element in pairs(AllElements) do
			if Element.Frame and Element.Name then
				local ElementName = string.lower(Element.Name)
				Element.Frame.Visible = string.find(ElementName, SearchText, 1, true) ~= nil
			end
		end
		for _, Tab in pairs(TabHolder:GetChildren()) do
			if Tab:IsA("TextButton") then
				local TabName = string.lower(Tab:FindFirstChild("Title") and Tab.Title.Text or "")
				Tab.Visible = string.find(TabName, SearchText, 1, true) ~= nil
			end
		end
	end

	local function CreateShootingStar(Parent)
		local StartX = math.random(10, 90) / 100
		local StartY = math.random(10, 60) / 100
		local EndX = StartX + math.random(15, 30) / 100
		local EndY = StartY + math.random(10, 20) / 100

		if EndX > 0.95 then EndX = 0.95 end
		if EndY > 0.9 then EndY = 0.9 end

		local Speed = math.random(8, 15) / 10

		local Star = Create("Frame", {
			Size = UDim2.new(0, math.random(60, 120), 0, 2),
			Position = UDim2.new(StartX, 0, StartY, 0),
			BackgroundColor3 = Color3.fromRGB(220 + math.random(0, 35), 220 + math.random(0, 35), 255),
			BorderSizePixel = 0,
			BackgroundTransparency = 1,
			ZIndex = -1,
			Rotation = math.random(25, 35),
			Parent = Parent
		}, {
			Create("UICorner", {CornerRadius = UDim.new(1, 0)}),
			Create("UIGradient", {
				Transparency = NumberSequence.new({
					NumberSequenceKeypoint.new(0, 1),
					NumberSequenceKeypoint.new(0.2, 0.2),
					NumberSequenceKeypoint.new(0.5, 0),
					NumberSequenceKeypoint.new(0.8, 0.2),
					NumberSequenceKeypoint.new(1, 1)
				})
			})
		})

		table.insert(ShootingStars, Star)

		task.spawn(function()
			task.wait(math.random(3, 8))
			TweenService:Create(Star, TweenInfo.new(Speed, Enum.EasingStyle.Sine, Enum.EasingDirection.In), {
				BackgroundTransparency = 0,
				Position = UDim2.new(EndX, 0, EndY, 0)
			}):Play()

			task.wait(Speed)
			TweenService:Create(Star, TweenInfo.new(0.3, Enum.EasingStyle.Sine), {
				BackgroundTransparency = 1
			}):Play()

			task.wait(0.3)
			Star:Destroy()

			for i, v in ipairs(ShootingStars) do
				if v == Star then
					table.remove(ShootingStars, i)
					break
				end
			end
		end)

		return Star
	end

	local function CreateStar(Parent)
		local StarSize = math.random(1, 4)
		local Star = Create("Frame", {
			Size = UDim2.new(0, StarSize, 0, StarSize),
			Position = UDim2.new(math.random(5, 95)/100, 0, math.random(5, 95)/100, 0),
			BackgroundColor3 = Color3.fromRGB(200 + math.random(0, 55), 200 + math.random(0, 55), 255),
			BorderSizePixel = 0,
			BackgroundTransparency = math.random(20, 70) / 100,
			ZIndex = -1,
			Parent = Parent
		}, {
			Create("UICorner", {CornerRadius = UDim.new(1, 0)})
		})

		table.insert(Stars, Star)

		local TwinkleInfo = {
			Star = Star,
			Speed = math.random(20, 40) / 10,
			Phase = math.random() * math.pi * 2,
			BaseTransparency = Star.BackgroundTransparency,
			Amplitude = math.random(20, 40) / 100
		}
		table.insert(StarTwinkles, TwinkleInfo)

		return Star
	end

	local function UpdateStarTwinkles()
		for _, info in ipairs(StarTwinkles) do
			if info.Star and info.Star.Parent then
				local newTrans = info.BaseTransparency + math.sin(tick() * info.Speed + info.Phase) * info.Amplitude
				info.Star.BackgroundTransparency = math.clamp(newTrans, 0.1, 0.9)
			end
		end
	end

	local function UpdateElementScaling()
		local windowWidth = MainWindow.AbsoluteSize.X
		local windowHeight = MainWindow.AbsoluteSize.Y
		local baseWidth = 615
		local baseHeight = 344

		local widthScale = math.clamp(windowWidth / baseWidth, 0.6, 1.5)
		local heightScale = math.clamp(windowHeight / baseHeight, 0.6, 1.5)
		local scaleFactor = (widthScale + heightScale) / 2

		if not Minimized then
			local baseTitleSize = 21
			local newTitleSize = math.floor(baseTitleSize * math.min(scaleFactor, 1.2))
			WindowName.TextSize = math.max(12, math.min(28, newTitleSize))
		else
			WindowName.TextSize = 21
		end

		for _, element in ipairs(AllElements) do
			if element.Frame then
				local labels = element.Frame:GetDescendants()
				for _, label in ipairs(labels) do
					if label:IsA("TextLabel") or label:IsA("TextButton") then
						local baseSize = 15
						if label.Name == "Title" then baseSize = 16
						elseif label.Name == "Content" then baseSize = 14
						elseif label.Name == "Value" then baseSize = 13 end

						local newSize = math.floor(baseSize * scaleFactor)
						label.TextSize = math.max(10, math.min(24, newSize))
						if label.Name == "Value" then
							label.TextWrapped = false
						else
							label.TextWrapped = true
						end
					end
				end
			end
		end

		for _, label in ipairs(SectionLabels) do
			if label and label.Parent then
				local baseSize = 14
				local newSize = math.floor(baseSize * scaleFactor)
				label.TextSize = math.max(11, math.min(22, newSize))
				label.TextWrapped = true
			end
		end

		for _, tab in ipairs(TabHolder:GetChildren()) do
			if tab:IsA("TextButton") then
				local title = tab:FindFirstChild("Title")
				if title then
					local newSize = math.floor(14 * scaleFactor)
					title.TextSize = math.max(11, math.min(20, newSize))
					title.TextWrapped = true
				end
			end
		end

		SearchBox.TextSize = math.floor(14 * scaleFactor)

		local titleWidth = WindowName.TextBounds.X
		local availableWidth = MainWindow.AbsoluteSize.X - 150
		if not Minimized and titleWidth > availableWidth and availableWidth > 50 then
			WindowName.TextWrapped = true
		else
			WindowName.TextWrapped = false
		end
	end

	local function animateTyping(newText)
		if titleTyping then return end
		titleTyping = true

		local currentText = WindowName.Text
		local deleteDelay = 0.03
		local typeDelay = 0.05

		for i = #currentText, 0, -1 do
			WindowName.Text = string.sub(currentText, 1, i)
			task.wait(deleteDelay)
		end

		task.wait(0.1)

		for i = 1, #newText do
			WindowName.Text = string.sub(newText, 1, i)
			task.wait(typeDelay)
		end

		titleTyping = false

		if Minimized then
			UpdateMinimizedWidth()
		end
	end

	local function MinimizeWindow()
		if isMinimizing then return end

			local targetSize = UDim2.new(0, 615, 0, 344)
			MainWindow:TweenSize(targetSize, Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.25, true)

			task.wait(0.2)

			WindowTopBarLine.Visible = true
			ResizeHandle.Visible = not FullScreen
			MinimizeBtn.Ico.Image = "rbxassetid://7072719338"
			MinimizeBtn.Ico.Rotation = 0
		else
			WindowTopBarLine.Visible = false
			WindowStuff.Visible = false
			ResizeHandle.Visible = false
			WindowName.TextSize = 21

			local titleWidth = WindowName.TextBounds.X
			local padding = 170
			local newWidth = math.clamp(titleWidth + padding, 200, 500)
			local targetSize = UDim2.new(0, newWidth, 0, 55)

			MainWindow:TweenSize(targetSize, Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.25, true)
			MinimizeBtn.Ico.Image = "rbxassetid://7072720870"
			MinimizeBtn.Ico.Rotation = 180
		end
		Minimized = not Minimized

		task.wait(0.25)
		isMinimizing = false
	end

				Text = Text or "Text"
				Content = Content or "Content"

				local ParagraphFrame = AddThemeObject(SetChildren(SetProps(MakeElement("RoundFrame", Color3.fromRGB(255, 255, 255), 0, 8), {

					ParagraphFrame.Size = UDim2.new(1, 0, 0, contentHeight + 42)

				local contentHeight = ParagraphFrame.Content.TextBounds.Y
				if contentHeight < 20 then contentHeight = 20 end
				ParagraphFrame.Content.Size = UDim2.new(1, -32, 0, contentHeight)
				ParagraphFrame.Size = UDim2.new(1, 0, 0, contentHeight + 42)

						end
					end
				end
				return ParagraphFunction
			end

					SliderDrag.Value.Text = tostring(self.Value) .. " " .. SliderConfig.ValueName

				function Dropdown:Refresh(Options, Delete)
					if Delete then
						for _,v in pairs(Dropdown.Buttons) do
							v:Destroy()
						end
						table.clear(Dropdown.Options)
						table.clear(Dropdown.Buttons)
						table.clear(Dropdown.Selected)
					end
					Dropdown.Options = Options
					AddOptions(Dropdown.Options)
					DropdownFrame.F.Selected.Text = GetSelectedText()
				end

				function Dropdown:Add(Value)
					if not table.find(Dropdown.Options, Value) then return end
					if not table.find(Dropdown.Selected, Value) then
						table.insert(Dropdown.Selected, Value)
						DropdownFrame.F.Selected.Text = GetSelectedText()
						DropdownConfig.Callback(Dropdown.Selected)
					end
				end

				function Dropdown:Remove(Value)
					local index = table.find(Dropdown.Selected, Value)
					if index then
						table.remove(Dropdown.Selected, index)
						DropdownFrame.F.Selected.Text = GetSelectedText()
						DropdownConfig.Callback(Dropdown.Selected)
					end
				end

				function Dropdown:Set(Values)
					if type(Values) ~= "table" then return end
					table.clear(Dropdown.Selected)
					for _, value in ipairs(Values) do

							table.insert(Dropdown.Selected, value)
						end
					end

						local isSel = table.find(Dropdown.Selected, option) ~= nil
						btn.BackgroundTransparency = isSel and 0.2 or 1
						if btn.CheckIcon then
							btn.CheckIcon.Size = UDim2.new(0, isSel and 20 or 0, 0, isSel and 20 or 0)
							btn.CheckIcon.ImageTransparency = isSel and 0 or 1
						end
					end

					if SelectAllBtn then
						SelectAllBtn.DisplayName.Text = #Dropdown.Selected == #Dropdown.Options and "Deselect All" or "Select All"
					end

					DropdownFrame.F.Selected.Text = GetSelectedText()
					DropdownConfig.Callback(Dropdown.Selected)
				end

			function ElementFunction:AddColorpicker(ColorpickerConfig)
				ColorpickerConfig = ColorpickerConfig or {}
				ColorpickerConfig.Name = ColorpickerConfig.Name or "Colorpicker"
				ColorpickerConfig.Default = ColorpickerConfig.Default or Color3.fromRGB(255,255,255)
				ColorpickerConfig.Callback = ColorpickerConfig.Callback or function() end
				ColorpickerConfig.Flag = ColorpickerConfig.Flag or nil
				ColorpickerConfig.Save = ColorpickerConfig.Save or false

				local ColorH, ColorS, ColorV = 1, 1, 1
				local Colorpicker = {Value = ColorpickerConfig.Default, Toggled = false, Type = "Colorpicker", Save = ColorpickerConfig.Save}

				local ColorSelection = Create("ImageLabel", {
					Size = UDim2.new(0, 20, 0, 20),
					Position = UDim2.new(select(3, Color3.toHSV(Colorpicker.Value))),
					ScaleType = Enum.ScaleType.Fit,
					AnchorPoint = Vector2.new(0.5, 0.5),
					BackgroundTransparency = 1,
					Image = "http://www.roblox.com/asset/?id=4805639000"
				})

				local HueSelection = Create("ImageLabel", {
					Size = UDim2.new(0, 20, 0, 20),
					Position = UDim2.new(0.5, 0, 1 - select(1, Color3.toHSV(Colorpicker.Value))),
					ScaleType = Enum.ScaleType.Fit,
					AnchorPoint = Vector2.new(0.5, 0.5),
					BackgroundTransparency = 1,
					Image = "http://www.roblox.com/asset/?id=4805639000"
				})

				local Color = Create("ImageLabel", {
					Size = UDim2.new(1, -28, 1, 0),
					Visible = false,
					Image = "rbxassetid://4155801252"
				}, {
					Create("UICorner", {CornerRadius = UDim.new(0, 7)}),
					ColorSelection
				})

				local Hue = Create("Frame", {
					Size = UDim2.new(0, 22, 1, 0),
					Position = UDim2.new(1, -22, 0, 0),
					Visible = false
				}, {
					Create("UIGradient", {Rotation = 270, Color = ColorSequence.new{ColorSequenceKeypoint.new(0.00, Color3.fromRGB(255, 0, 4)), ColorSequenceKeypoint.new(0.20, Color3.fromRGB(234, 255, 0)), ColorSequenceKeypoint.new(0.40, Color3.fromRGB(21, 255, 0)), ColorSequenceKeypoint.new(0.60, Color3.fromRGB(0, 255, 255)), ColorSequenceKeypoint.new(0.80, Color3.fromRGB(0, 17, 255)), ColorSequenceKeypoint.new(0.90, Color3.fromRGB(255, 0, 251)), ColorSequenceKeypoint.new(1.00, Color3.fromRGB(255, 0, 4))},}),
					Create("UICorner", {CornerRadius = UDim.new(0, 7)}),
					HueSelection
				})

				local ColorpickerContainer = Create("Frame", {
					Position = UDim2.new(0, 0, 0, 36),
					Size = UDim2.new(1, 0, 1, -36),
					BackgroundTransparency = 1,
					ClipsDescendants = true
				}, {
					Hue,
					Color,
					Create("UIPadding", {
						PaddingLeft = UDim.new(0, 38),
						PaddingRight = UDim.new(0, 38),
						PaddingBottom = UDim.new(0, 12),
						PaddingTop = UDim.new(0, 20)
					})
				})

				local Click = SetProps(MakeElement("Button"), {
					Size = UDim2.new(1, 0, 1, 0)
				})

				local ColorpickerBox = AddThemeObject(SetChildren(SetProps(MakeElement("RoundFrame", Color3.fromRGB(255, 255, 255), 0, 6), {
					Size = UDim2.new(0, 28, 0, 28),
					Position = UDim2.new(1, -16, 0.5, 0),
					AnchorPoint = Vector2.new(1, 0.5)
				}), {
					AddThemeObject(MakeElement("Stroke", OrionLib.Themes.Default.Stroke, 1.2), "Stroke")
				}), "Main")

				local ColorpickerFrame = AddThemeObject(SetChildren(SetProps(MakeElement("RoundFrame", Color3.fromRGB(255, 255, 255), 0, 8), {
					Size = UDim2.new(1, 0, 0, 42),
					Parent = ItemParent
				}), {
					SetProps(SetChildren(MakeElement("TFrame"), {
						AddThemeObject(SetProps(MakeElement("Label", ColorpickerConfig.Name, 15), {
							Size = UDim2.new(1, -16, 1, 0),
							Position = UDim2.new(0, 16, 0, 0),
							Font = Enum.Font.GothamBold,
							Name = "Content",
							TextTruncate = Enum.TextTruncate.AtEnd,
							TextWrapped = true
						}), "Text"),
						ColorpickerBox,
						Click,
						AddThemeObject(SetProps(MakeElement("Frame"), {
							Size = UDim2.new(1, 0, 0, 1.5),
							Position = UDim2.new(0, 0, 1, -1.5),
							Name = "Line",
							Visible = false
						}), "Stroke"),
					}), {
						Size = UDim2.new(1, 0, 0, 42),
						ClipsDescendants = true,
						Name = "F"
					}),
					ColorpickerContainer,
					AddThemeObject(MakeElement("Stroke", OrionLib.Themes.Default.Stroke, 1.2), "Stroke"),
				}), "Second")

				AddConnection(Click.MouseButton1Click, function()
					Colorpicker.Toggled = not Colorpicker.Toggled
					TweenService:Create(ColorpickerFrame, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = Colorpicker.Toggled and UDim2.new(1, 0, 0, 170) or UDim2.new(1, 0, 0, 42)}):Play()
					Color.Visible = Colorpicker.Toggled
					Hue.Visible = Colorpicker.Toggled
					ColorpickerFrame.F.Line.Visible = Colorpicker.Toggled
					TweenService:Create(ColorpickerFrame.F.Ico, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
						Rotation = Colorpicker.Toggled and 90 or 0
					}):Play()
				end)

				local function UpdateColorPicker()
					ColorpickerBox.BackgroundColor3 = Color3.fromHSV(ColorH, ColorS, ColorV)
					Color.BackgroundColor3 = Color3.fromHSV(ColorH, 1, 1)
					Colorpicker:Set(ColorpickerBox.BackgroundColor3)
					ColorpickerConfig.Callback(ColorpickerBox.BackgroundColor3)
					SaveCfg(game.GameId)
				end

				ColorH = 1 - (math.clamp(HueSelection.AbsolutePosition.Y - Hue.AbsolutePosition.Y, 0, Hue.AbsoluteSize.Y) / Hue.AbsoluteSize.Y)
				ColorS = (math.clamp(ColorSelection.AbsolutePosition.X - Color.AbsolutePosition.X, 0, Color.AbsoluteSize.X) / Color.AbsoluteSize.X)
				ColorV = 1 - (math.clamp(ColorSelection.AbsolutePosition.Y - Color.AbsolutePosition.Y, 0, Color.AbsoluteSize.Y) / Color.AbsoluteSize.Y)

				local Dragging, DragInput, MousePos, FramePos = false

				AddConnection(Color.InputBegan, function(Input)
					if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
						Dragging = true
						MousePos = Input.Position
						FramePos = ColorSelection.Position

						AddConnection(Input.Changed, function()
							if Input.UserInputState == Enum.UserInputState.End then
								Dragging = false
								FocusDrag = nil
							end
						end)
					end
				end)

				AddConnection(Color.InputChanged, function(Input)
					if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch and not FocusDrag then
						DragInput = Input
						FocusDrag = DragInput
					end
				end)

				AddConnection(UserInputService.InputChanged, function(Input)
					if Input == DragInput and Dragging and Input == FocusDrag then
						local ColorX = (math.clamp(DragInput.Position.X - Color.AbsolutePosition.X, 0, Color.AbsoluteSize.X) / Color.AbsoluteSize.X)
						local ColorY = (math.clamp(Input.Position.Y - Color.AbsolutePosition.Y, 0, Color.AbsoluteSize.Y) / Color.AbsoluteSize.Y)
						ColorSelection.Position = UDim2.new(ColorX, 0, ColorY, 0)
						ColorS = ColorX
						ColorV = 1 - ColorY
						UpdateColorPicker()
					end
				end)

				local Dragging_1, DragInput_1, MousePos_1, FramePos_1 = false

				AddConnection(Hue.InputBegan, function(Input)
					if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
						Dragging_1 = true
						MousePos_1 = Input.Position
						FramePos_1 = HueSelection.Position

						AddConnection(Input.Changed, function()
							if Input.UserInputState == Enum.UserInputState.End then
								Dragging_1 = false
								FocusDrag = nil
							end
						end)
					end
				end)
				AddConnection(Hue.InputChanged, function(Input)
					if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch and not FocusDrag then
						DragInput_1 = Input
						FocusDrag = DragInput_1
					end
				end)

				AddConnection(UserInputService.InputChanged, function(Input)
					if Input == DragInput_1 and Dragging_1 and DragInput_1 == FocusDrag then
						local HueY = (math.clamp(Input.Position.Y - Hue.AbsolutePosition.Y, 0, Hue.AbsoluteSize.Y) / Hue.AbsoluteSize.Y)
						HueSelection.Position = UDim2.new(0.5, 0, HueY, 0)
						ColorH = 1 - HueY
						UpdateColorPicker()
					end
				end)

				function Colorpicker:Set(Value)
					Colorpicker.Value = Value
					ColorpickerBox.BackgroundColor3 = Colorpicker.Value
					ColorpickerConfig.Callback(Colorpicker.Value)
				end

				Colorpicker:Set(Colorpicker.Value)
				if ColorpickerConfig.Flag then
					OrionLib.Flags[ColorpickerConfig.Flag] = Colorpicker
				end
				return Colorpicker
			end

			return ElementFunction
		end

		local ElementFunction = {}

		function ElementFunction:AddSection(SectionConfig)
			SectionConfig.Name = SectionConfig.Name or "Section"

			local SectionFrame = SetChildren(SetProps(MakeElement("TFrame"), {
				Size = UDim2.new(1, 0, 0, 26),
				Parent = Container
			}), {
				AddThemeObject(SetProps(MakeElement("Label", SectionConfig.Name, 14), {
					Size = UDim2.new(1, -12, 0, 16),
					Position = UDim2.new(0, 0, 0, 3),
					Font = Enum.Font.GothamSemibold,
					Name = "SectionTitle"
				}), "TextDark"),
				SetChildren(SetProps(MakeElement("TFrame"), {
					AnchorPoint = Vector2.new(0, 0),
					Size = UDim2.new(1, 0, 1, -24),
					Position = UDim2.new(0, 0, 0, 23),
					Name = "Holder"
				}), {
					MakeElement("List", 0, 6)
				}),
			})

			local sectionLabel = SectionFrame:FindFirstChild("SectionTitle")
			if sectionLabel then
				table.insert(SectionLabels, sectionLabel)
			end

			AddConnection(SectionFrame.Holder.UIListLayout:GetPropertyChangedSignal("AbsoluteContentSize"), function()
				SectionFrame.Size = UDim2.new(1, 0, 0, SectionFrame.Holder.UIListLayout.AbsoluteContentSize.Y + 31)
				SectionFrame.Holder.Size = UDim2.new(1, 0, 0, SectionFrame.Holder.UIListLayout.AbsoluteContentSize.Y)
			end)

			local SectionFunction = {}
			for i, v in next, GetElements(SectionFrame.Holder) do
				SectionFunction[i] = v
			end
			return SectionFunction
		end

		for i, v in next, GetElements(Container) do
			ElementFunction[i] = v
		end

		return ElementFunction
	end

	return TabFunction
