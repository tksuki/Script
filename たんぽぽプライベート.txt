local function spawntoy(name, cframe, vector3)
    local inv = SpawnedInToys
    local toy = SpawnToyRemoteFunction:InvokeServer(table.unpack({
        [1] = name, [2] = cframe, [3] = vector3 or Vector3.zero
    }))
    return inv and inv[name]
end

local function FWD(parent, part, timeout)
    return parent:FindFirstChild(part) or parent:WaitForChild(part, timeout or 3)
end

local function GetPlot()
    local plots = service.Workspace:FindFirstChild("Plots")
    if not plots then return end
    local folder = PLOT_MAP[Config.SelectedPlot]
    if folder then return plots:FindFirstChild(folder) end
end

local function isTargetable(player)
    if not player or player == localPlayer then return false end
    if Config.WhitelistKickall then
        local isFriend = false
        pcall(function() isFriend = localPlayer:IsFriendsWith(player.UserId) end)
        if isFriend then return false end
    end
    return true
end

local function IsPlayerInsideSafeZone(player)
    return player:FindFirstChild("InPlot") and player.InPlot.Value
end

local function ragdolallwhite(p)
    if not p or not p.UserId or not localPlayer then return false end
    local Friends = false
    pcall(function() Friends = localPlayer:IsFriendsWith(p.UserId) end)
    return Friends
end

local function whitelistBlob(Player)
    if not Config.WhitelistFriendsBLOB then return false end
    if not Player or not Player.UserId then return false end
    if Player == localPlayer then return true end
    return localPlayer:IsFriendsWith(Player.UserId)
end

local function Whiteall(player)
    if not Config.bloballWhite then return false end
    if player == localPlayer then return true end
    if localPlayer:IsFriendsWith(player.UserId) then return true end
    if Config.bloballWhitePlr[player.Name] then return true end
    return false
end

local function getBlobman()
    local inv = SpawnedInToys
    local v = inv and inv:FindFirstChild("CreatureBlobman")
    if not v then
        for _, p in ipairs(PlotItems:GetChildren()) do
            local m = p:FindFirstChild("CreatureBlobman")
            if m and m:FindFirstChild("PlayerValue") and m.PlayerValue.Value == localPlayer.Name then
                v = m
                break
            end
        end
    end
    return v
end

local function spawnBLob2()
    local char = GetCharacter()
    local tFolder = service.Workspace:WaitForChild(localPlayer.Name .. "SpawnedInToys")
    local existing = tFolder:FindFirstChild("CreatureBlobman")
    if existing then return existing end
    task.spawn(function()
        pcall(function() spawntoy("CreatureBlobman", char.Head.CFrame, Vector3.new(0, 0, 0)) end)
    end)
    local blobman
    for _ = 1, 50 do
        blobman = tFolder:FindFirstChild("CreatureBlobman")
        if blobman then break end
        task.wait(0.05)
    end
    return blobman
end

local function sitOnBlobman(blobman)
    local char = localPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChild("Humanoid")
    if not root or not humanoid then return end
    if not blobman then
        spawnBLob2()
        blobman = getBlobman()
    end
    if blobman then
        local seat = blobman:FindFirstChildOfClass("VehicleSeat") or blobman:FindFirstChild("VehicleSeat")
        if seat and humanoid.SeatPart ~= seat then
            seat:Sit(humanoid)
        end
    end
end

local FlowerESP = Tabs.FlowerTab:AddLeftGroupbox("ESP")
do
    local ESPConnections = {}
    local PCLD_ESP_CON = nil
    local RainbowConnection = nil
    local ESPBool = { ESPEnabled = false, PCLDESP = false }

    local function RainbowColor(offset)
        local r = math.sin(offset)*0.5+0.5
        local g = math.sin(offset+2.094)*0.5+0.5
        local b = math.sin(offset+4.188)*0.5+0.5
        return Color3.new(math.max(0.25,r), math.max(0.25,g), math.max(0.25,b))
    end
    local function createESP(plr)
        if plr == localPlayer then return end
        local function onAdd(char)
            local Head = char:FindFirstChild("Head")
            if not Head or Head:FindFirstChild("BillboardGui") then return end
            local bg = Instance.new("BillboardGui")
            bg.Adornee = Head bg.Size = UDim2.new(0,50,0,50) bg.AlwaysOnTop = true
            bg.StudsOffset = Vector3.new(0,3,0) bg.Parent = Head
            local il = Instance.new("ImageLabel")
            il.BackgroundTransparency = 1 il.Size = UDim2.new(0,50,0,50)
            il.Position = UDim2.new(0.5,-25,0,0)
            il.Image = "https://www.roblox.com/headshot-thumbnail/image?userId="..plr.UserId.."&width=420&height=420&format=png"
            il.Parent = bg
            local tl = Instance.new("TextLabel")
            tl.BackgroundTransparency = 1 tl.Size = UDim2.new(1,0,0,32)
            tl.Position = UDim2.new(0,0,1,-32)
            tl.Text = plr.Name.." ("..plr.DisplayName..")"
            tl.TextColor3 = Color3.new(1,1,1) tl.TextStrokeTransparency = 0
            tl.TextScaled = true tl.Font = Enum.Font.GothamBold
            tl.Parent = bg
        end
        if plr.Character then onAdd(plr.Character) end
        table.insert(ESPConnections, plr.CharacterAdded:Connect(onAdd))
    end
    local function clearESP()
        for _, c in ipairs(ESPConnections) do c:Disconnect() end
        ESPConnections = {}
        for _, plr in ipairs(service.Players:GetPlayers()) do
            if plr.Character and plr.Character:FindFirstChild("Head") then
                local g = plr.Character.Head:FindFirstChild("BillboardGui")
                if g then g:Destroy() end
            end
        end
    end
    local function AddPCLD(obj)
        if obj.Name=="PlayerCharacterLocationDetector" and not obj:FindFirstChild("PCLD_ESP") then
            local a = Instance.new("BoxHandleAdornment")
            a.Name="PCLD_ESP" a.Size=obj.Size a.AlwaysOnTop=true a.ZIndex=10
            a.Color3=Color3.fromRGB(255,0,0) a.Transparency=0.8 a.Adornee=obj a.Parent=obj
            local s = Instance.new("SelectionBox")
            s.Name="PCLD_Edge" s.Adornee=obj s.LineThickness=0.04
            s.Color3=Color3.fromRGB(255,0,0) s.SurfaceTransparency=1 s.Transparency=0.45 s.Parent=obj
        end
    end
    local function RemovePCLD(obj)
        if obj.Name=="PlayerCharacterLocationDetector" then
            local e = obj:FindFirstChild("PCLD_ESP") if e then e:Destroy() end
            local e2 = obj:FindFirstChild("PCLD_Edge") if e2 then e2:Destroy() end
        end
    end
    local function UpdateRainbow()
        if not ESPBool.PCLDESP then
            if RainbowConnection then RainbowConnection:Disconnect() RainbowConnection=nil end
            return
        end
        local c = RainbowColor(tick()*2.5)
        for _, v in pairs(service.Workspace:GetChildren()) do
            if v.Name=="PlayerCharacterLocationDetector" then
                local e = v:FindFirstChild("PCLD_Edge") if e then e.Color3=c end
                local e2 = v:FindFirstChild("PCLD_ESP") if e2 then e2.Color3=c end
            end
        end
    end

    FlowerESP:AddToggle("PlayerESPToggle", {
        Text = "Player ESP", Default = false,
        Callback = function(v)
            ESPBool.ESPEnabled = v
            if v then
                for _, plr in pairs(service.Players:GetPlayers()) do
                    if plr ~= localPlayer then createESP(plr) end
                end
                if not ESPConnections.NewPlayer then
                    ESPConnections.NewPlayer = service.Players.PlayerAdded:Connect(function(plr)
                        if plr ~= localPlayer then createESP(plr) end
                    end)
                end
            else
                clearESP()
                if ESPConnections.NewPlayer then ESPConnections.NewPlayer:Disconnect() ESPConnections.NewPlayer=nil end
            end
        end,
    })
    FlowerESP:AddToggle("PCLDESPToggle", {
        Text = "PCLD ESP", Default = false,
        Callback = function(v)
            ESPBool.PCLDESP = v
            if v then
                for _, obj in pairs(service.Workspace:GetChildren()) do AddPCLD(obj) end
                PCLD_ESP_CON = service.Workspace.ChildAdded:Connect(AddPCLD)
                if not RainbowConnection then RainbowConnection = service.RunService.Heartbeat:Connect(UpdateRainbow) end
            else
                for _, obj in pairs(service.Workspace:GetChildren()) do RemovePCLD(obj) end
                if PCLD_ESP_CON then PCLD_ESP_CON:Disconnect() PCLD_ESP_CON=nil end
                if RainbowConnection then RainbowConnection:Disconnect() RainbowConnection=nil end
            end
        end,
    })
    FlowerESP:AddButton({
        Text = "Reset ESP",
        Func = function()
            if ESPBool.ESPEnabled then
                ESPBool.ESPEnabled=false clearESP()
                if ESPConnections.NewPlayer then ESPConnections.NewPlayer:Disconnect() ESPConnections.NewPlayer=nil end
            end
            if ESPBool.PCLDESP then
                ESPBool.PCLDESP=false
                for _, v in pairs(service.Workspace:GetChildren()) do RemovePCLD(v) end
                if PCLD_ESP_CON then PCLD_ESP_CON:Disconnect() PCLD_ESP_CON=nil end
                if RainbowConnection then RainbowConnection:Disconnect() RainbowConnection=nil end
            end
        end,
    })

local FlowerCornerESP = Tabs.FlowerTab:AddLeftGroupbox("Corner ESP")
do
    local RB_ESP = false
    local rb_espObjects = {}
    local rb_espUpdateConn = nil
    local rb_globalHue = 0
    local rb_pulseTime = 0
    local rb_playerAddedConn = nil
    local rb_playerRemovingConn = nil

    local rb_espGui = Instance.new("ScreenGui")
    rb_espGui.Name = "TanpopoCornerESP"
    rb_espGui.ResetOnSpawn = false
    rb_espGui.IgnoreGuiInset = true
    rb_espGui.Parent = game:GetService("CoreGui")

    local function RB_getTorsoCenter(char)
        local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if torso then return torso.Position
        elseif hrp then return hrp.Position + Vector3.new(0, 0.5, 0) end
        return nil
    end
    local function RB_getPlayerHeight(char)
        local head = char:FindFirstChild("Head")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
        local lowerTorso = char:FindFirstChild("LowerTorso")
        if not head or not hrp then return 4 end
        local topY = head.Position.Y + head.Size.Y/2
        local bottomY
        if lowerTorso then bottomY = lowerTorso.Position.Y - lowerTorso.Size.Y/2
        elseif torso then bottomY = torso.Position.Y - torso.Size.Y/2
        else bottomY = hrp.Position.Y - 2 end
        return topY - bottomY
    end
    local function RB_AddESP(p)
        if p == localPlayer or rb_espObjects[p] then return end
        if not p.Character then return end
        local container = Instance.new("Frame")
        container.BackgroundTransparency = 1
        container.Size = UDim2.new(0, 80, 0, 80)
        container.Visible = false
        container.Parent = rb_espGui
        local thickness = 2
        local cornerSize = 8
        local function createCornerLine(anchorPoint, position, size)
            local line = Instance.new("Frame")
            line.BackgroundColor3 = Color3.fromHSV(0, 1, 1)
            line.BorderSizePixel = 0
            line.AnchorPoint = anchorPoint
            line.Position = position
            line.Size = size
            line.Parent = container
            return line
        end
        createCornerLine(Vector2.new(0, 0), UDim2.new(0, 0, 0, 0), UDim2.new(0, cornerSize, 0, thickness))
        createCornerLine(Vector2.new(0, 0), UDim2.new(0, 0, 0, 0), UDim2.new(0, thickness, 0, cornerSize))
        createCornerLine(Vector2.new(1, 0), UDim2.new(1, 0, 0, 0), UDim2.new(0, cornerSize, 0, thickness))
        createCornerLine(Vector2.new(1, 0), UDim2.new(1, 0, 0, 0), UDim2.new(0, thickness, 0, cornerSize))
        createCornerLine(Vector2.new(0, 1), UDim2.new(0, 0, 1, 0), UDim2.new(0, cornerSize, 0, thickness))
        createCornerLine(Vector2.new(0, 1), UDim2.new(0, 0, 1, 0), UDim2.new(0, thickness, 0, cornerSize))
        createCornerLine(Vector2.new(1, 1), UDim2.new(1, 0, 1, 0), UDim2.new(0, cornerSize, 0, thickness))
        createCornerLine(Vector2.new(1, 1), UDim2.new(1, 0, 1, 0), UDim2.new(0, thickness, 0, cornerSize))
        rb_espObjects[p] = { container = container, lines = container:GetChildren(), rotation = 0, player = p }
    end
    local function RB_RemoveESP(p)
        if rb_espObjects[p] then rb_espObjects[p].container:Destroy() rb_espObjects[p] = nil end
    end
    local function RB_RemoveAllESP()
        for p, data in pairs(rb_espObjects) do
            if data and data.container then data.container:Destroy() end
        end
        rb_espObjects = {}
    end
    local function RB_AddAllESP()
        for _, p in service.Players:GetPlayers() do
            if p ~= localPlayer then RB_AddESP(p) end
        end
    end
    local function RB_UpdateESP()
        if not RB_ESP then return end
        rb_globalHue = (rb_globalHue + 0.004) % 1
        local currentColor = Color3.fromHSV(rb_globalHue, 1, 1)
        rb_pulseTime = rb_pulseTime + 0.025
        local pulseFactor = 0.7 + 0.3 * math.sin(rb_pulseTime)
        for _, player in service.Players:GetPlayers() do
            if player ~= localPlayer then
                if not rb_espObjects[player] then RB_AddESP(player) end
                local data = rb_espObjects[player]
                if data then
                    local container = data.container
                    local char = player.Character
                    if not char then container.Visible = false
                    else
                        local torsoPos = RB_getTorsoCenter(char)
                        if not torsoPos then container.Visible = false
                        else
                            local height = RB_getPlayerHeight(char)
                            local dist = (service.Workspace.CurrentCamera.CFrame.Position - torsoPos).Magnitude
                            local vec, onScreen = service.Workspace.CurrentCamera:WorldToViewportPoint(torsoPos)
                            if onScreen then
                                local heightOnScreen = (height / dist) * (service.Workspace.CurrentCamera.FieldOfView * 0.3)
                                local baseSize = heightOnScreen * 90
                                local finalSize = baseSize * pulseFactor
                                finalSize = math.clamp(finalSize, 25, 130)
                                if dist > 800 then container.Visible = false
                                else
                                    container.Size = UDim2.new(0, finalSize, 0, finalSize)
                                    container.Position = UDim2.new(0, vec.X - finalSize/2, 0, vec.Y - finalSize/2)
                                    container.Visible = true
                                    data.rotation = (data.rotation + 1.0) % 360
                                    container.Rotation = data.rotation
                                    for _, line in ipairs(data.lines) do
                                        if line:IsA("Frame") then line.BackgroundColor3 = currentColor end
                                    end
                                end
                            else container.Visible = false end
                        end
                    end
                end
            end
        end
    end
    FlowerCornerESP:AddToggle("RainbowCornerESP", {
        Text = "Rainbow Corner ESP", Default = false,
        Callback = function(Value)
            RB_ESP = Value
            if Value then
                RB_RemoveAllESP()
                RB_AddAllESP()
                if rb_espUpdateConn then rb_espUpdateConn:Disconnect() end
                rb_espUpdateConn = service.RunService.RenderStepped:Connect(RB_UpdateESP)
                if rb_playerAddedConn then rb_playerAddedConn:Disconnect() end
                if rb_playerRemovingConn then rb_playerRemovingConn:Disconnect() end
                rb_playerAddedConn = service.Players.PlayerAdded:Connect(function(p)
                    if RB_ESP then task.wait(0.5) RB_AddESP(p) end
                end)
                rb_playerRemovingConn = service.Players.PlayerRemoving:Connect(RB_RemoveESP)

local InsertService = service.InsertService
local function WipeOriginal(bh)
    if not bh then return end
    for _, desc in ipairs(bh:GetDescendants()) do
        pcall(function()
            if desc:IsA("BasePart") or desc:IsA("MeshPart") or desc:IsA("UnionOperation") then
                desc.Transparency = 1 desc.CastShadow = false
            elseif desc:IsA("ParticleEmitter") or desc:IsA("Trail") or desc:IsA("Beam") or desc:IsA("Smoke") or desc:IsA("Fire") or desc:IsA("Sparkles") then
                desc.Enabled = false
            elseif desc:IsA("PointLight") or desc:IsA("SpotLight") or desc:IsA("SurfaceLight") then
                desc.Enabled = false
            elseif desc:IsA("Decal") or desc:IsA("Texture") then
                desc.Transparency = 1
            elseif desc:IsA("BillboardGui") or desc:IsA("SurfaceGui") then
                desc.Enabled = false
            end
        end)
    end
end
local function RestoreOriginal(bh)
    if not bh then return end
    for _, desc in ipairs(bh:GetDescendants()) do
        pcall(function()
            if desc:IsA("BasePart") or desc:IsA("MeshPart") or desc:IsA("UnionOperation") then
                desc.Transparency = 0
            elseif desc:IsA("ParticleEmitter") or desc:IsA("Trail") or desc:IsA("Beam") or desc:IsA("Smoke") or desc:IsA("Fire") or desc:IsA("Sparkles") then
                desc.Enabled = true
            elseif desc:IsA("PointLight") or desc:IsA("SpotLight") or desc:IsA("SurfaceLight") then
                desc.Enabled = true
            elseif desc:IsA("Decal") or desc:IsA("Texture") then
                desc.Transparency = 0
            elseif desc:IsA("BillboardGui") or desc:IsA("SurfaceGui") then
                desc.Enabled = true
            end
        end)
    end
end
local function Cleanup(bh)
    if not bh then return end
    local data = CosmicBlackhole.Active[bh]
    if not data then return end
    if data.Anim then pcall(function() data.Anim:Disconnect() end) end
    if data.Model then pcall(function() data.Model:Destroy() end) end
    pcall(function() RestoreOriginal(bh) end)
    CosmicBlackhole.Active[bh] = nil
end
local function LoadCustomModel(assetId)
    if not assetId then return nil end
    local cleanId = tonumber((assetId:gsub("rbxassetid://", "")))
    if not cleanId then return nil end
    local ok1, res1 = pcall(function() return game:GetObjects(assetId)[1] end)
    if ok1 and res1 then return res1 end
    local ok2, res2 = pcall(function() return InsertService:LoadAsset(cleanId) end)
    if ok2 and res2 then
        local children = res2:GetChildren()
        if #children > 0 then return children[1] end
        return res2
    end
    return nil
end
local function Build(bh)
    if not bh or not bh.Parent then return end
    if CosmicBlackhole.Active[bh] then return end
    local hole = bh:FindFirstChild("Hole") or bh:FindFirstChildWhichIsA("BasePart")
    if not hole then return end
    local option = CosmicBlackhole.Options[CosmicBlackhole.Selected]
    if not option then return end
    local model = LoadCustomModel(option.Id)
    if not model then return end
    WipeOriginal(bh)
    local pos = hole.Position
    local data = { Time = 0, Hole = hole, Anim = nil, Model = nil, SpinSpeed = 0, TargetSpin = 0.8, Scale = 0, TargetScale = option.Scale }
    model.Parent = bh
    model.Name = "CosmicMesh"
    for _, desc in ipairs(model:GetDescendants()) do
        pcall(function()
            if desc:IsA("AnimationController") or desc:IsA("Animator") or desc:IsA("Animation") then desc:Destroy() end
        end)
    end
    for _, part in ipairs(model:GetDescendants()) do
        pcall(function()
            if part:IsA("BasePart") or part:IsA("MeshPart") then
                part.Anchored = true part.CanCollide = false part.CastShadow = false
                part.Size = part.Size * 0.01
            end
            if part:IsA("Smoke") or part:IsA("ParticleEmitter") then
                local n = part.Name:lower()
                if n:find("smog") or n:find("fog") or n:find("mist") or n:find("haze") or n:find("cloud") then part:Destroy() end
            end
        end)
    end
    pcall(function()
        if model.PrimaryPart then
            local offset = pos - model.PrimaryPart.Position
            model:SetPrimaryPartCFrame(model.PrimaryPart.CFrame + offset)
        else
            local firstPart = nil
            for _, part in ipairs(model:GetDescendants()) do
                if part:IsA("BasePart") then firstPart = part break end
            end
            if firstPart then
                local offset = pos - firstPart.Position
                for _, part in ipairs(model:GetDescendants()) do
                    if part:IsA("BasePart") then part.CFrame = part.CFrame + offset end
                end
            end
        end
    end)
    data.Model = model
    data.Anim = service.RunService.Heartbeat:Connect(function(dt)
        if not bh or not bh.Parent then Cleanup(bh) return end
        local currentPos = hole.Position
        data.Time = data.Time + dt * data.SpinSpeed
        if data.Model and data.Model.Parent then
            if data.Model:IsA("Model") then
                if data.Model.PrimaryPart then
                    local cf = CFrame.new(currentPos) * CFrame.Angles(0, data.Time, 0)
                    pcall(function() data.Model:SetPrimaryPartCFrame(cf) end)
                end
                for _, part in ipairs(data.Model:GetDescendants()) do
                    if part:IsA("BasePart") or part:IsA("MeshPart") then
                        pcall(function()
                            if not part:GetAttribute("BaseSize") then part:SetAttribute("BaseSize", part.Size / 0.01) end
                            local baseSize = part:GetAttribute("BaseSize")
                            part.Size = baseSize * data.Scale
                        end)
                    end
                end
            elseif data.Model:IsA("BasePart") then
                pcall(function()
                    data.Model.CFrame = CFrame.new(currentPos) * CFrame.Angles(0, data.Time, 0)
                    if not data.Model:GetAttribute("BaseSize") then data.Model:SetAttribute("BaseSize", data.Model.Size / 0.01) end
                    data.Model.Size = data.Model:GetAttribute("BaseSize") * data.Scale
                end)
            end
        end
    end)
    bh.AncestryChanged:Connect(function()
        if not bh.Parent then Cleanup(bh) end
    end)
    CosmicBlackhole.Active[bh] = data
end
local function StartSystem()
    if CosmicBlackhole.Connection then return end
    CosmicBlackhole.Connection = service.Workspace.ChildAdded:Connect(function(obj)
        if not CosmicBlackhole.Enabled then return end
        if obj.Name:lower():find("blackhole") then task.wait(0.2) Build(obj) end
    end)
end

local function CanAuraTarget(targetPlayer)
    if not targetPlayer or targetPlayer == localPlayer then return false end
    if not targetPlayer.Character then return false end
    if AuraConfig.WhitelistFriends and localPlayer:IsFriendsWith(targetPlayer.UserId) then
        return false
    end
    local hum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

local function AuraIsFriend(p)
    if not p or not p.UserId then return false end
    local ok, r = pcall(function() return localPlayer:IsFriendsWith(p.UserId) end)
    return ok and r or false
end

local function AuraCFP(parent, part) return parent:FindFirstChild(part) ~= nil end
local function AuraGetMagnitude(Part1, Part2) return (Part1.Position - Part2.Position).Magnitude end

local function AuraGetPlayersInRadius()
    local hrp = HRP()
    local targets = {}
    local radius = 30
    for _, p in pairs(service.Players:GetPlayers()) do
        if p == localPlayer then continue end
        if AuraConfig.WhitelistFriends and AuraIsFriend(p) then continue end
        local char = p.Character
        if not char then continue end
        local tHrp = char:FindFirstChild("HumanoidRootPart")

                table.insert(targets, {
                    Player = p, Character = char, Root = tHrp,
                    Head = char:FindFirstChild("Head"), Hum = hum, Distance = dist,
                })
            end
        end
    end
    return targets
end

local function AuraGetCircularPositions(count, centerX, centerZ, baseRadius, minDistance)
    if count == 0 then return {} end
    if count == 1 then return {{X = centerX, Z = centerZ}} end
    local positions = {}
    local usedPositions = {}
    baseRadius = baseRadius or 8
    minDistance = minDistance or 2
    local adjustedRadius = math.max(baseRadius, count * 0.8 + 2)
    local maxAttempts = 50
    for i = 1, count do
        local angle = (i - 1) * (2 * math.pi / count) + math.random(-0.03, 0.03)
        local radiusOffset = math.random(-0.3, 0.3)
        local actualRadius = adjustedRadius + radiusOffset
        local x = centerX + actualRadius * math.cos(angle)
        local z = centerZ + actualRadius * math.sin(angle)
        local attempts = 0
        local validPosition = false
        while not validPosition and attempts < maxAttempts do
            validPosition = true
            local pos = Vector3.new(x, 0, z)
            for _, usedPos in ipairs(usedPositions) do
                local dist = (pos - usedPos).Magnitude
                if dist < minDistance then
                    validPosition = false
                    local pushAngle = math.atan2(z - centerZ, x - centerX)
                    local pushRadius = (actualRadius + minDistance - dist) * 0.3
                    x = x + pushRadius * math.cos(pushAngle + math.random(-0.2, 0.2))
                    z = z + pushRadius * math.sin(pushAngle + math.random(-0.2, 0.2))
                    break
                end
            end
            attempts = attempts + 1
            if attempts > maxAttempts then
                x = centerX + (adjustedRadius + count * 0.3) * math.cos(i * 1.3 + 0.5)
                z = centerZ + (adjustedRadius + count * 0.3) * math.sin(i * 1.3 + 0.5)
                break
            end
        end
        table.insert(usedPositions, Vector3.new(x, 0, z))
        table.insert(positions, {X = x, Z = z})
    end
    return positions
end

    DestroyGrabLine:FireServer(root)
    service.RunService.RenderStepped:Wait()

    if target.Player.IsHeld and target.Player.IsHeld.Value then
        for _, v in pairs(target.Character:GetChildren()) do
            if v:IsA("BasePart") then
                if fixedPosition then v.CFrame = CFrame.new(fixedPosition)

        end

            end
        end
        RagdollRemote:FireServer(root, 1)
    end
    service.RunService.RenderStepped:Wait()
    DestroyGrabLine:FireServer(root)
end

                                    local targetPos = Vector3.new(0,0,0)
                                    root.CFrame = CFrame.new(targetPos)
                                    if AuraConfig.teleportBodie[player] and AuraConfig.teleportBodie[player].Parent then AuraConfig.teleportBodie[player]:Destroy() end
                                    local bp = Instance.new("BodyPosition")
                                    bp.Name = "TeleportAuraBP"
                                    bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                                    bp.P = 50000
                                    bp.D = 1250
                                    bp.Position = targetPos
                                    bp.Parent = root
                                    AuraConfig.teleportBodie[player] = bp
                                    if AuraConfig.teleportGyro[player] and AuraConfig.teleportGyro[player].Parent then AuraConfig.teleportGyro[player]:Destroy() end
                                    local bg = Instance.new("BodyGyro")
                                    bg.Name = "TeleportAuraBG"
                                    bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                                    bg.CFrame = root.CFrame
                                    bg.Parent = root
                                    AuraConfig.teleportGyro[player] = bg
                                end
                            end
                        end
                    end
                end
            end)
            coroutine.resume(AuraConfig.teleportCoro)
        else
            if AuraConfig.teleportCoro then coroutine.close(AuraConfig.teleportCoro) AuraConfig.teleportCoro = nil end
            for _, bp in pairs(AuraConfig.teleportBodie) do if bp and bp.Parent then bp:Destroy() end end
            for _, bg in pairs(AuraConfig.teleportGyro) do if bg and bg.Parent then bg:Destroy() end end
            AuraConfig.teleportBodie, AuraConfig.teleportGyro = {}, {}
        end
    end
})

kickaura:AddToggle("KickAuraT", {
    Text = "KickAura (spam)",
    Default = false,
    Callback = function(value)
        local hrp = HRP()
        if value then
            if AuraConfig.spamkickaura then return end
            AuraConfig.spamkickaura = true
            if AuraConfig.LineLagEnabled then
                if AuraConfig.spamkickauralag then
                    AuraConfig.spamkickauralag = false
                    AuraConfig.Contuuti:cancel("LineLagThread")
                end
                AuraConfig.spamkickauralag = true
                AuraConfig.Contuuti:cancel("LineLagThread")
                AuraConfig.Contuuti["LineLagThread"] = task.spawn(function()
                    while AuraConfig.spamkickauralag do
                        local spawnLocation = service.Workspace:FindFirstChild("SpawnLocation") or service.Workspace:FindFirstChild("Spawn") or hrp
                        if spawnLocation then
                            local randomX = math.random(-1e9, 1e9)
                            local randomZ = math.random(-1e9, 1e9)
                            local directions = {
                                CFrame.new(randomX, 0, randomZ), CFrame.new(-randomX, 0, -randomZ),
                                CFrame.new(randomX, 0, -randomZ), CFrame.new(-randomX, 0, randomZ),
                                CFrame.new(randomX * 0.5, 0, randomZ * 0.5), CFrame.new(-randomX * 0.5, 0, -randomZ * 0.5),
                            }
                            for _, pos in pairs(directions) do
                                pcall(function() CreateGrabLine:FireServer(spawnLocation, pos) end)
                            end
                        end
                        task.wait()
                    end
                end)
            end
            if AuraConfig.DoRagdoll and AuraConfig.UsePallete then
                local pallete = SpawnedInToys:FindFirstChild("RagdollPalete")
                if not pallete then AuraSpawnPallete() end
            end
            AuraConfig.Contuuti:cancel("LoopGrabKickThread")
            AuraConfig.Contuuti["LoopGrabKickThread"] = task.spawn(function()
                local targetData = {}
                local circularPositions = {}
                local lastUpdateTime = 0
                local positionUpdateInterval = 0.5
                while AuraConfig.spamkickaura do
                    service.RunService.RenderStepped:Wait()
                    local targets = AuraGetPlayersInRadius()
                    local currentTargetNames = {}
                    for _, t in pairs(targets) do currentTargetNames[t.Player.Name] = true end
                    for name, data in pairs(targetData) do
                        if not currentTargetNames[name] then
                            if data.BodyPos then data.BodyPos:Destroy() end
                            targetData[name] = nil
                        end
                    end
                    local currentTime = tick()
                    if currentTime - lastUpdateTime > positionUpdateInterval or #targets ~= #targetData then
                        local myPos = hrp.Position
                        local count = #targets
                        if count > 0 then
                            local Radius = 30
                            local MinDistance = 2
                            circularPositions = AuraGetCircularPositions(count, myPos.X, myPos.Z, Radius * 0.25, MinDistance)
                            lastUpdateTime = currentTime
                        end
                    end
                    local targetIndex = 0
                    for _, target in pairs(targets) do
                        targetIndex = targetIndex + 1
                        local name = target.Player.Name
                        if not targetData[name] then
                            targetData[name] = {Player = target.Player, Root = target.Root, Head = target.Head, Hum = target.Hum, Character = target.Character, BodyPos = nil}
                        end
                        local data = targetData[name]
                        data.Root = target.Root data.Head = target.Head data.Hum = target.Hum data.Character = target.Character
                        local pos = circularPositions[targetIndex]
                        local fixedPos = nil
                        if pos then fixedPos = Vector3.new(pos.X, hrp.Position.Y + 15, pos.Z) end
                        AuraKickTarget(target, fixedPos)
                    end
                    task.wait(0.05)
                end
                for name, data in pairs(targetData) do
                    if data.BodyPos then data.BodyPos:Destroy() end
                    targetData[name] = nil
                end
                if AuraConfig.LineLagEnabled then AuraConfig.spamkickauralag = false AuraConfig.Contuuti:cancel("LineLagThread") end
            end)
        else
            if not AuraConfig.spamkickaura then return end
            AuraConfig.spamkickaura = false
            AuraConfig.Contuuti:cancel("LoopGrabKickThread")
            if AuraConfig.LineLagEnabled then AuraConfig.spamkickauralag = false AuraConfig.Contuuti:cancel("LineLagThread") end
            for _, player in pairs(service.Players:GetPlayers()) do
                if player ~= localPlayer then
                    local char = player.Character
                    if char then
                        local root = char:FindFirstChild("HumanoidRootPart")
                        if root then
                            for _, v in pairs(root:GetChildren()) do if v:IsA("BodyPosition") then v:Destroy() end end
                        end
                        local hum = char:FindFirstChild("Humanoid")
                        if hum then hum.PlatformStand = false end
                    end
                end
            end
        end
    end
})
