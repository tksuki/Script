local function sendChat(message)
    local RBXGeneral = TextChatService.TextChannels:FindFirstChild("RBXGeneral")

    if RBXGeneral then
        pcall(function()
            RBXGeneral:SendAsync(message)
        end)
    else
        pcall(function()
            ReplicatedStorage.DefaultChatSystemChatEvents.SayMessageRequest:FireServer(message, "All")
        end)
    end
end

local function startLoop()
    if loopRunning then return end
    loopRunning = true

    task.spawn(function()
        for _, msg in ipairs(chatMessages) do
            if not loopRunning then return end
            sendChat(msg)
            task.wait(chatInterval)
        end
    end)
end

local function startLoopRepeat()
    if nearLoopRunning then return end
    nearLoopRunning = true

    task.spawn(function()
        while true do
            if not nearLoopRunning then return end
            for _, msg in ipairs(chatMessages) do
                if not nearLoopRunning then break end
                sendChat(msg)
                task.wait(chatInterval)
            end
            task.wait(0.5)
        end
    end)
end

local function stopAllLoops()

            if not targetLoopRunning then return end
            sendChat(targetPlayer.DisplayName .. " " .. msg)
            task.wait(targetInterval)
        end
    end)
end

local function getNearbyPlayerNames()
    local result = {}
    local selfChar = LocalPlayer.Character

    if not selfChar then return result end

    local selfRoot = selfChar:FindFirstChild("HumanoidRootPart")
    if not selfRoot then return result end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end

        local char = player.Character
        if not char then continue end

        local root     = char:FindFirstChild("HumanoidRootPart")
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not root then continue end

        local alive = humanoid and humanoid.Health > 0
        if not alive then continue end

        local dist = (root.Position - selfRoot.Position).Magnitude
        if dist <= nearRange then
            table.insert(result, { display = player.DisplayName, dist = dist })
        end
    end

    table.sort(result, function(a, b) return a.dist < b.dist end)

    local names = {}
    for _, v in ipairs(result) do
        table.insert(names, v.display)
    end
    return names
end

local nearOnceRunning    = false

local function sendToNearOnce()
    if nearOnceRunning then return end
    nearOnceRunning = true

    task.spawn(function()
        local msgIdx = 1
        while true do
            if not nearOnceRunning then return end

            local nearby = getNearbyPlayerNames()
            if #nearby == 0 then
                Library:Notify("近くにプレイヤーがいません", 2)
                nearOnceRunning = false
                return
            end

            local sent = false
            local currentMsg = chatMessages[msgIdx]
            if not currentMsg then break end

            for _, name in ipairs(nearby) do
                if not nearOnceRunning then break end
                sendChat(name .. " " .. currentMsg)
                task.wait(nearInterval)
                sent = true
            end

            if not sent then
                nearOnceRunning = false
                return
            end

            msgIdx = msgIdx + 1
            if msgIdx > #chatMessages then
                nearOnceRunning = false
                return
            end
        end
    end)
end

local function startNearLoop()
    if nearLoopRunning2 then return end
    nearLoopRunning2 = true

    task.spawn(function()
        local n = 1
        while true do
            while true do
                if not nearLoopRunning2 then return end
                local nearby = getNearbyPlayerNames()
                if #nearby > 0 then break end
                task.wait(0.5)
            end

            local nearby  = getNearbyPlayerNames()
            local msg     = chatMessages[n]

            for _, name in ipairs(nearby) do
                if not nearLoopRunning2 then break end
                sendChat(name .. " " .. msg)
                task.wait(nearInterval)
            end

            n = n + 1
            if n > #chatMessages then n = 1 end
        end
    end)
end

local function stopNearLoops()
    nearLoopRunning2 = false
    nearOnceRunning  = false
end

local function isOtherPlayerPart(part)
    if not (part and part:IsA("BasePart")) then
        return true
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char    = player.Character
            local backpack = player:FindFirstChild("Backpack")

            if char and part:IsDescendantOf(char) then return true end
            if backpack and part:IsDescendantOf(backpack) then return true end
        end
    end

    local ok, owner = pcall(function() return part:GetNetworkOwner() end)
    if not ok then return false end
    if owner == nil then return false end
    return owner ~= LocalPlayer
end

local function isDicePart(part)
    if not part or not part.IsA(part, "BasePart") or part.Anchored then return false end

    local selfChar = LocalPlayer.Character
    if selfChar and part:IsDescendantOf(selfChar) then return false end
    if isOtherPlayerPart(part) then return false end

    local name       = part.Name
    local parentName = part.Parent and part.Parent.Name or ""

    return (name:find("Dice") or parentName:find("Dice") or name:find("Toy") or parentName:find("Toy")) and true or false
end

local function setupDicePart(part)
    if not (part and part:IsA("BasePart")) then return end

    part.CanCollide = false

    if not dicePool.Orig[part] then
        dicePool.Orig[part] = {
            Color      = part.Color,
            CFrame     = part.CFrame,
            CanCollide = part.CanCollide
        }
    end

    local df = part:FindFirstChild("DiceForce") or Instance.new("BodyVelocity")
    df.Name      = "DiceForce"
    df.MaxForce  = Vector3.new(1, 1, 1) * 10000000
    df.P         = 12500
    df.Parent    = part

    local dg = part:FindFirstChild("DiceGyro") or Instance.new("BodyGyro")
    dg.Name      = "DiceGyro"
    dg.MaxTorque = Vector3.new(1, 1, 1) * 10000000
    dg.P         = 12500
    dg.Parent    = part
end

local function cleanupDicePart(part)
    if not part then return end

    local df = part:FindFirstChild("DiceForce")
    local dg = part:FindFirstChild("DiceGyro")

    if df then df:Destroy() end
    if dg then dg:Destroy() end
    part.CanCollide = true
end

local function pruneInvalidDice()
    for i = #dicePool.All, 1, -1 do
        local part = dicePool.All[i]
        if not part or not part.Parent or not part:IsDescendantOf(Workspace) or isOtherPlayerPart(part) then
            dicePool.HoveringMap[part] = nil
            dicePool.ColorMap[part]    = nil
            cleanupDicePart(part)
            table.remove(dicePool.All, i)
        end
    end
end

local function collectDice()
    pruneInvalidDice()

    local selfChar = LocalPlayer.Character
    if not selfChar then return end

    local root = selfChar:FindFirstChild("HumanoidRootPart") or selfChar:FindFirstChild("Torso")
    if not root then return end

    for _, part in ipairs(Workspace:GetPartBoundsInRadius(root.Position, 120)) do
        if #dicePool.All >= diceState.maxDice then break end
        if isDicePart(part) and not dicePool.HoveringMap[part] then
            dicePool.HoveringMap[part] = true
            setupDicePart(part)
            table.insert(dicePool.All, part)
            if not dicePool.ColorMap[part] then
                dicePool.ColorMap[part] = Colors.DEFAULT
            end
        end
    end

    dicePool.Right = {}
    dicePool.Left  = {}
    for i, v in ipairs(dicePool.All) do
        if i % 2 == 1 then
            table.insert(dicePool.Right, v)
        else
            table.insert(dicePool.Left, v)
        end
    end
end

local function sphereOffset(index, total, radius, acc, speed)
    if total <= 0 then return Vector3.zero end

    local phi    = (1 + 2.23606797749979) / 2
    local theta  = 2 * math.pi * index / phi
    local z      = 1 - 2 * (index - 0.5) / total
    local sinZ   = math.sin(math.acos(z))
    local x      = radius * sinZ * math.cos(theta)
    local y      = radius * sinZ * math.sin(theta)
    local zComp  = radius * z
    local rot    = speed or diceState.rotSpeed
    local cf     = CFrame.Angles(acc * rot * 0.75, acc * rot, acc * rot * 0.5)
    return cf:VectorToWorldSpace(Vector3.new(x, y, zComp))
end

local function disableAllDice()
    for _, v in ipairs(dicePool.All) do
        cleanupDicePart(v)
    end
    dicePool.All        = {}
    dicePool.Right      = {}
    dicePool.Left       = {}
    dicePool.HoveringMap = {}
    dicePool.ColorMap   = {}
    diceState.animating = false
    animation.Type      = "NONE"
end

RunService.Heartbeat:Connect(function(dt)
    if not diceState.enabled then return end

    diceState.acc = diceState.acc + dt
    collectDice()

    local selfChar = LocalPlayer.Character
    if not selfChar then return end

    local HumanoidRootPart = selfChar:FindFirstChild("HumanoidRootPart") or selfChar:FindFirstChild("Torso")
    if not HumanoidRootPart then return end

    if diceState.thirdPerson then
        Camera.CameraType = Enum.CameraType.Scriptable
        local camPos = HumanoidRootPart.Position
            - HumanoidRootPart.CFrame.LookVector * diceState.thirdPersonDist
            + Vector3.new(0, 4, 0)
        Camera.CFrame = CFrame.new(camPos, HumanoidRootPart.Position + Vector3.new(0, 2, 0))
    end

    local offset = Vector3.new(0, diceState.heightOffset, 0)
    local posRight  = (HumanoidRootPart.CFrame * CFrame.new( 8, 0,  0)).Position + offset
    local posLeft   = (HumanoidRootPart.CFrame * CFrame.new(-8, 0,  0)).Position + offset
    local posFront  = (HumanoidRootPart.CFrame * CFrame.new( 0, 0, -6)).Position + offset
    local posAbove  = (HumanoidRootPart.CFrame * CFrame.new( 0, 8,  0)).Position + offset

    if not diceState.animating then
        local r = diceState.radius

        for idx, part in ipairs(dicePool.Right) do
            if part and part.Parent then
                local off = sphereOffset(idx, #dicePool.Right, r, diceState.acc)
                local df  = part:FindFirstChild("DiceForce")
                local dg  = part:FindFirstChild("DiceGyro")
                if df then df.Velocity = (posRight + off - part.Position) * 20 end
                if dg then dg.CFrame   = CFrame.new(part.Position, posRight + off) * CFrame.Angles(diceState.acc * diceState.rotSpeed, 0, 0) end
            end
        end

        for i, v in ipairs(dicePool.Left) do
            if v and v.Parent then
                local off = sphereOffset(i, #dicePool.Left, r, diceState.acc)
                local df  = v:FindFirstChild("DiceForce")
                local dg  = v:FindFirstChild("DiceGyro")
                if df then df.Velocity = (posLeft + off - v.Position) * 20 end
                if dg then dg.CFrame   = CFrame.new(v.Position, posLeft + off) * CFrame.Angles(diceState.acc * diceState.rotSpeed, 0, 0) end
            end
        end
        return
    end

    animation.Timer = animation.Timer + dt

    if animation.Type == "MERGE_AND_SHOOT" then
        if animation.Timer <= 1.8 then
            local t = math.clamp(animation.Timer / 1.8, 0, 1)
            for _, part in ipairs(dicePool.Right) do dicePool.ColorMap[part] = Colors.RED:Lerp(Colors.PURPLE, t) end
            for _, v   in ipairs(dicePool.Left)  do dicePool.ColorMap[v]    = Colors.BLUE:Lerp(Colors.PURPLE, t) end

            local cRight = posRight:Lerp(posFront, t)
            local cLeft  = posLeft:Lerp(posFront, t)
            for idx, part in ipairs(dicePool.Right) do
                local off = sphereOffset(idx, #dicePool.Right, 4, diceState.acc, 6)
                local df  = part:FindFirstChild("DiceForce")
                if df then df.Velocity = (cRight + off - part.Position) * 22 end
            end
            for i, v in ipairs(dicePool.Left) do
                local off = sphereOffset(i, #dicePool.Left, 4, diceState.acc, 6)
                local df  = v:FindFirstChild("DiceForce")
                if df then df.Velocity = (cLeft + off - v.Position) * 22 end
            end
        elseif animation.Timer <= 3.5 then
            local t      = animation.Timer - 1.8
            local target = posFront + animation.FlyDir * 300 * t
            for i, v in ipairs(dicePool.All) do
                if v and v.Parent then
                    dicePool.ColorMap[v] = Colors.PURPLE
                    local off = sphereOffset(i, #dicePool.All, 5.5, diceState.acc, 12)
                    local df  = v:FindFirstChild("DiceForce")
                    if df then df.Velocity = (target + off - v.Position) * 40 end
                end
            end
        else
            diceState.animating = false
            animation.Type      = "NONE"
        end
        return
    end

    if animation.Type == "PURPLE_MERGE" then
        local t = math.clamp(animation.Timer / 1.5, 0, 1)
        for _, part in ipairs(dicePool.Right) do dicePool.ColorMap[part] = Colors.RED:Lerp(Colors.PURPLE,  t) end
        for _, v   in ipairs(dicePool.Left)  do dicePool.ColorMap[v]    = Colors.BLUE:Lerp(Colors.PURPLE, t) end

        local cRight = posRight:Lerp(posFront, t)
        local cLeft  = posLeft:Lerp(posFront, t)
        for idx, part in ipairs(dicePool.Right) do
            local off = sphereOffset(idx, #dicePool.Right, 4.5, diceState.acc)
            local df  = part:FindFirstChild("DiceForce")
            if df then df.Velocity = (cRight + off - part.Position) * 20 end
        end
        for i, v in ipairs(dicePool.Left) do
            local off = sphereOffset(i, #dicePool.Left, 4.5, diceState.acc)
            local df  = v:FindFirstChild("DiceForce")
            if df then df.Velocity = (cLeft + off - v.Position) * 20 end
        end
        return
    end

    if animation.Type == "BLUE_BURST" then
        if animation.Timer <= 2.5 then
            local angle  = animation.Timer / 2.5 * math.pi * 2
            local orbit  = HumanoidRootPart.Position + Vector3.new(math.cos(angle) * 10, 2 + diceState.heightOffset, math.sin(angle) * 10)
            for i, v in ipairs(dicePool.All) do
                if v and v.Parent then
                    local off = sphereOffset(i, #dicePool.All, 5, diceState.acc, 10)
                    local df  = v:FindFirstChild("DiceForce")
                    if df then df.Velocity = (orbit + off - v.Position) * 30 end
                end
            end
        elseif animation.Timer <= 4.5 then
            local t = animation.Timer - 2.5
            for _, v in ipairs(dicePool.All) do
                if v and v.Parent then
                    local dir = animation.BurstDirs[v] or Vector3.new(0, 1, 0)
                    local df  = v:FindFirstChild("DiceForce")
                    if df then df.Velocity = dir * (200 + t * 50) end
                end
            end
        else
            diceState.animating = false
            animation.Type      = "NONE"
        end
        return
    end

    if animation.Type == "RED_SHOOT" then
        if animation.Timer <= 2 then
            for i, v in ipairs(dicePool.All) do
                if v and v.Parent then
                    local off = sphereOffset(i, #dicePool.All, 4, diceState.acc, 8)
                    local df  = v:FindFirstChild("DiceForce")
                    if df then df.Velocity = (posFront + off - v.Position) * 25 end
                end
            end
        elseif animation.Timer <= 3.5 then
            local t      = animation.Timer - 2
            local target = posFront + animation.FlyDir * 250 * t
            for i, v in ipairs(dicePool.All) do
                if v and v.Parent then
                    local off = sphereOffset(i, #dicePool.All, 4, diceState.acc, 8)
                    local df  = v:FindFirstChild("DiceForce")
                    if df then df.Velocity = (target + off - v.Position) * 35 end
                end
            end
        else
            diceState.animating = false
            animation.Type      = "NONE"
        end
        return
    end

    if animation.Type == "BLACK_HOLE" then
        for i, v in ipairs(dicePool.All) do
            if v and v.Parent then
                local off = sphereOffset(i, #dicePool.All, 4.5, diceState.acc, 8)
                local df  = v:FindFirstChild("DiceForce")
                if df then df.Velocity = (posAbove + off - v.Position) * 25 end
            end
        end
        return
    end
end)

local espEnabled    = false
local espHighlights = {}
local espColor      = Color3.fromRGB(255, 0, 0)
local espFillAlpha  = 0.5
local espOutlineAlpha = 0.5

local function addESP(player)
    if player == LocalPlayer then return end
    if not player.Character then return end
    if espHighlights[player] then return end

    local hl = Instance.new("Highlight")
    hl.Name                = "ESP"
    hl.Adornee             = player.Character
    hl.FillColor           = espColor
    hl.FillTransparency    = espFillAlpha
    hl.OutlineColor        = espColor
    hl.OutlineTransparency = espOutlineAlpha
    hl.DepthMode           = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent              = game.CoreGui
    espHighlights[player]  = hl
end

local function removeESP(player)
    if not espHighlights[player] then return end
    espHighlights[player]:Destroy()
    espHighlights[player] = nil
end

local function updateAllESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if espEnabled then
            addESP(player)
        else
            removeESP(player)
        end
    end
end

        if not espEnabled then return end

    if espEnabled then addESP(player) end
end)

local targetGroup = Tabs.Target:AddLeftGroupbox("指定プレイヤー冷笑", "target")

local function buildPlayerList()
    local list = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            table.insert(list, player.DisplayName .. " (@" .. player.Name .. ")")
        end
    end
    table.sort(list, function(a, b) return a:lower() < b:lower() end)
    return list
end

targetGroup:AddDropdown("TargetPlayerDropdown", {
    Text     = "対象プレイヤー",
    Values   = buildPlayerList(),
    Default  = "",
    Callback = function(val)
        if not val or val == "" then return end
        local name = val:match("@(.+)%)")
        if name then
            targetPlayer = Players:FindFirstChild(name)
        end
    end
})

Players.PlayerAdded:Connect(function()
    if Options.TargetPlayerDropdown then
        Options.TargetPlayerDropdown:SetValues(buildPlayerList())
    end
end)

Players.PlayerRemoving:Connect(function()
    if Options.TargetPlayerDropdown then
        Options.TargetPlayerDropdown:SetValues(buildPlayerList())
    end
end)
