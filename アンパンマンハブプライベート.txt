function isTargetable(player)

lineLagThread = nil
lineLagEnabled = false

function startLineLag()
    if lineLagEnabled then return end
    lineLagEnabled = true
    lineLagThread = coroutine.create(function()
        if not GrabEvents then return end
        local createLine = GrabEvents:FindFirstChild("CreateGrabLine")
        if not createLine then return end
        while lineLagEnabled do
            local spawnLocation = Workspace:FindFirstChild("SpawnLocation") or Workspace:FindFirstChild("Spawn") or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart"))
            if spawnLocation then
                local randomX = math.random(-9e9, 9e9)
                local randomZ = math.random(-9e9, 9e9)
                local directions = {
                    CFrame.new(randomX, 0, randomZ),
                    CFrame.new(-randomX, 0, -randomZ),
                    CFrame.new(randomX, 0, -randomZ),
                    CFrame.new(-randomX, 0, randomZ),
                }
                for _, pos in pairs(directions) do
                    pcall(function() createLine:FireServer(spawnLocation, pos) end)
                end
            end
            task.wait()
        end
    end)
    coroutine.resume(lineLagThread)
end

function stopLineLag()
    if not lineLagEnabled then return end
    lineLagEnabled = false
    if lineLagThread then coroutine.close(lineLagThread) lineLagThread = nil end
end

local function toy_spawn_gucci(name, cframe, vector)
    if not megaSpawnToyRF then return nil end

    local InOwnerPlot = LocalPlayer:FindFirstChild("InOwnedPlot")
    local CanSpawn = LocalPlayer:FindFirstChild("CanSpawnToy")

        task.wait(0.01)
    end
    task.spawn(function() megaSpawnToyRF:InvokeServer(name, cframe, vector or Vector3.new()) end)
    local BackPack = Workspace:FindFirstChild(LocalPlayer.Name .. 'SpawnedInToys')
    if not BackPack then return nil end
    local SpawnedToy = nil
    local connection
    connection = BackPack.ChildAdded:Connect(function(toy)
        if toy.Name == name and toy:IsA("Model") then
            SpawnedToy = toy
            connection:Disconnect()
        end
    end)
    local startTick = tick()
    while not SpawnedToy do
        if tick() - startTick > 2 then
            if connection then connection:Disconnect() end
            return nil
        end
        task.wait(0.01)
    end
    return SpawnedToy

local function antiInputOnWorkspaceChild(v)

end

                if v.Name == "HoldPart" then
                    local isHeld = false
                    local owner = v:FindFirstChild("PartOwner")
                    if not owner then
                        local parentModel = v.Parent
                        if parentModel then
                            owner = parentModel:FindFirstChild("PartOwner")
                        end
                    end
                    if owner and owner.Value ~= "" then
                        isHeld = true
                    end

                    if isHeld then
                        local drop = v:FindFirstChild("DropItemRemoteFunction")
                        if drop then
                            t[#t + 1] = {
                                Part = v,
                                Drop = drop,
                                ForceDrop = true
                            }
                        end
                    elseif not antiInputSeenHolds[v] then
                        local hold = v:FindFirstChild("HoldItemRemoteFunction")
                        local drop = v:FindFirstChild("DropItemRemoteFunction")
                        if hold and drop then
                            t[#t + 1] = {
                                Part = v,
                                Hold = hold,
                                Drop = drop,
                                ForceDrop = false
                            }
                        end
                    end
                end
            end
        end
    end)
    for h in pairs(antiInputSeenHolds) do
        if not h.Parent then
            antiInputSeenHolds[h] = nil
        end
    end
    return t
end

local function TeleportToMouse()
    local character = LocalPlayer.Character
    if not character then
        Library:Notify("Teleport", "Character not found", 2)
        return
    end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then
        Library:Notify("Teleport", "HRP not found", 2)
        return
    end
    
    local mouse = LocalPlayer:GetMouse()
    if not mouse then
        Library:Notify("Teleport", "Mouse not available", 2)
        return
    end
    
    local targetPos = mouse.Hit.p
    if not targetPos then
        Library:Notify("Teleport", "Invalid target position", 2)
        return
    end

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
    raycastParams.FilterDescendantsInstances = {character}
    local rayOrigin = targetPos + Vector3.new(0, 100, 0)
    local rayDirection = Vector3.new(0, -200, 0)
    local rayResult = workspace:Raycast(rayOrigin, rayDirection, raycastParams)
    local finalPos
    if rayResult then
        finalPos = rayResult.Position + Vector3.new(0, 3, 0)
    else
        finalPos = targetPos + Vector3.new(0, 3, 0)
    end
    
    hrp.CFrame = CFrame.new(finalPos)
    Library:Notify("Teleport", "Teleported to mouse position", 2)
end

-- ★★★ ターゲット判定を拡張（名前だけでなく、子孫に特定の名前を含むかもチェック） ★★★
local function IsPCLDTarget(obj)
    if not obj:IsA("BasePart") then return false end
    local name = string.lower(obj.Name)
    -- 一般的な名前
    if name == "playercharacterlocationdetector" or 
       name == "partesp" or 
       name == "pcld" or 
       name == "locationdetector" or
       name == "esp" or
       name == "detector" or
       name == "highlight" or
       name == "selectionbox" then
        return true
    end
    -- 子孫に特定の名前があるか（例：PartOwner など）
    for _, child in ipairs(obj:GetDescendants()) do
        if child:IsA("BasePart") and string.lower(child.Name):find("detector") then
            return true
        end
        if child.Name == "PartOwner" then
            return true
        end
    end
    return false
end

-- ★★★ メインレンダリングループ（変数名修正） ★★★
local renderConn = RunService.RenderStepped:Connect(function(dt)
    if not CFG.playerLocationVisualEnabled then
        for part, esp in pairs(esps) do
            esp.container.Visible = false
            esp.boxPart.Transparency = 1
            esp.selBox.Visible = false
        end
        return
    end

    local t = tick()

    local espColor
    if CFG.locationESPGradientEnabled then
        local alpha = 0.5 + 0.5 * math.sin(t * CFG.locationESPGradientSpeed * math.pi)
        espColor = CFG.locationESPGradientColor1:Lerp(CFG.locationESPGradientColor2, alpha)
    else
        espColor = CFG.locationESPColor
    end

    for part, esp in pairs(esps) do
        if not part or not part.Parent then
            removePart(part)
            continue
        end

        local partPos = part.Position
        if math.abs(partPos.X) < 0.01 and math.abs(partPos.Y) < 0.01 and math.abs(partPos.Z) < 0.01 then
            esp.container.Visible = false
            esp.boxPart.Transparency = 1
            esp.selBox.Visible = false
            continue
        end

        if CFG.locationESPSmoothingEnabled then
            local alpha = 1 - math.exp(-CFG.locationESPSmoothingSpeed * dt)
            esp.estimatedCFrame = esp.estimatedCFrame:Lerp(part.CFrame, alpha)
        else
            esp.estimatedCFrame = part.CFrame
        end

        local espPartColor = CFG.locationESPGradientEnabled and espColor or CFG.locationESPPartColor

        if CFG.locationESPPartEnabled then
            esp.boxPart.CFrame = esp.estimatedCFrame
            esp.boxPart.Size = part.Size
            local mat = Enum.Material[CFG.locationESPPartMaterial]
            if mat then esp.boxPart.Material = mat end
            esp.boxPart.Color = espPartColor
            esp.boxPart.Transparency = CFG.locationESPPartTransparency

            if CFG.locationESPBoxOutlineEnabled then
                esp.selBox.Visible = true
                esp.selBox.Color3 = CFG.locationESPBoxOutlineColor
                esp.selBox.LineThickness = CFG.locationESPBoxOutlineThickness
            else
                esp.selBox.Visible = false
            end
        else
            esp.boxPart.Transparency = 1
            esp.selBox.Visible = false
        end

        -- ★★★ カメラ変数を正しく参照 ★★★
        local cam = workspace.CurrentCamera
        if not cam then continue end
        local wPos = esp.estimatedCFrame.Position
        local screenPos, onScreen = cam:WorldToViewportPoint(wPos)

        if onScreen then
            esp.container.Visible = true
            esp.container.Position = UDim2.fromOffset(screenPos.X, screenPos.Y)

            if CFG.locationESPCornerEnabled then
                esp.cornersContainer.Visible = true
                esp.rotAngle = (esp.rotAngle + dt * CFG.locationESPRotationSpeed) % 360
                esp.cornersContainer.Rotation = esp.rotAngle

                local range = CFG.locationESPMaxRadius - CFG.locationESPMinRadius
                local breath = CFG.locationESPMinRadius + range * (0.5 + 0.5 * math.sin(t * CFG.locationESPExpandSpeed))
                local length = CFG.locationESPLength
                local thick = CFG.locationESPThickness

                esp.corners[1].Position = UDim2.fromOffset(-breath, -breath)
                esp.corners[2].Position = UDim2.fromOffset(breath, -breath)
                esp.corners[3].Position = UDim2.fromOffset(breath, breath)
                esp.corners[4].Position = UDim2.fromOffset(-breath, breath)

                for _, cf in ipairs(esp.corners) do
                    cf.Size = UDim2.fromOffset(length, length)
                    for _, child in ipairs(cf:GetChildren()) do
                        if child.Name == "Horiz" then
                            child.Size = UDim2.fromOffset(length, thick)
                            child.BackgroundColor3 = espColor
                            child.BackgroundTransparency = 0
                            local s = child:FindFirstChild("OutlineStroke")
                            if s then
                                s.Enabled = CFG.locationESPCornerOutlineEnabled
                                s.Color = CFG.locationESPCornerOutlineColor
                                s.Thickness = CFG.locationESPCornerOutlineThickness
                            end
                        elseif child.Name == "Vert" then
                            child.Size = UDim2.fromOffset(thick, length)
                            child.BackgroundColor3 = espColor
                            child.BackgroundTransparency = 0
                            local s = child:FindFirstChild("OutlineStroke")
                            if s then
                                s.Enabled = CFG.locationESPCornerOutlineEnabled
                                s.Color = CFG.locationESPCornerOutlineColor
                                s.Thickness = CFG.locationESPCornerOutlineThickness
                            end
                        elseif child.Name == "Dot" then
                            child.BackgroundColor3 = espColor
                            child.BackgroundTransparency = 0
                            child.Size = UDim2.fromOffset(thick + 2, thick + 2)
                            child.Position = UDim2.fromOffset(-1, -1)
                        end
                    end
                end
            else
                esp.cornersContainer.Visible = false
            end
        else
            esp.container.Visible = false
        end
    end
end)
table.insert(visualConnections, renderConn)

-- クリーンアップ関数
function CleanupPCLDESP()
    for part, esp in pairs(esps) do
        pcall(function() esp.sg:Destroy() end)
        pcall(function() esp.boxPart:Destroy() end)
        pcall(function() esp.selBox:Destroy() end)
    end
    table.clear(esps)
    table.clear(activeParts)
    for _, c in ipairs(visualConnections) do
        pcall(function() c:Disconnect() end)
    end
    table.clear(visualConnections)
    CFG.playerLocationVisualEnabled = false
end

        if kickLoopEnabled and not selectedTarget then

            statusLabel.Text = "Error: No target selected"
            toggleButton.Text = "Loop Kick: OFF"
            toggleButton.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
            return
        end

        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChild("Humanoid")
        local seat = hum and hum.SeatPart

        if kickLoopEnabled and (not seat or seat.Parent.Name ~= "CreatureBlobman") then
            kickLoopEnabled = false
            statusLabel.Text = "Error: Not riding a Blobman"
            toggleButton.Text = "Loop Kick: OFF"
            toggleButton.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
            return
        end

        if kickLoopEnabled then
            toggleButton.Text = "Loop Kick: ON"
            toggleButton.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
            statusLabel.Text = "Kick loop started"
        else
            toggleButton.Text = "Loop Kick: OFF"
            toggleButton.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
            statusLabel.Text = "Kick loop stopped"
            if loopThread then
                task.cancel(loopThread)
                loopThread = nil
            end
            return
        end

        -- ループ開始
        loopThread = task.spawn(function()
            local RS = game:GetService("ReplicatedStorage")
            local GE = RS:WaitForChild("GrabEvents")

            local blob = seat.Parent
            local blobRoot = blob:FindFirstChild("HumanoidRootPart") or blob.PrimaryPart
            local scriptObj = blob:FindFirstChild("BlobmanSeatAndOwnerScript")

            local CG = scriptObj and scriptObj:FindFirstChild("CreatureGrab")
            local CD = scriptObj and scriptObj:FindFirstChild("CreatureDrop")

            local R_Det = blob:FindFirstChild("RightDetector")
            local R_Weld = R_Det and (R_Det:FindFirstChild("RightWeld") or R_Det:FindFirstChildWhichIsA("Weld"))

            local SNO = GE:WaitForChild("SetNetworkOwner")
            local CreateGrabLine = GE:WaitForChild("CreateGrabLine")
            local DestroyGrabLine = GE:WaitForChild("DestroyGrabLine")

            local SavedPos = blobRoot.CFrame
            local packetCount = 0
            local wasDead = false
            local lockPos = SavedPos * CFrame.new(0, 19, 0)

            local function TeleportToTarget()
                if selectedTarget and selectedTarget.Character then
                    local tRoot = selectedTarget.Character:FindFirstChild("HumanoidRootPart")
                    if tRoot and blobRoot then
                        blobRoot.CFrame = tRoot.CFrame
                        blobRoot.Velocity = Vector3.zero
                        if CG and R_Det then CG:FireServer(R_Det, tRoot, R_Weld) end
                        if CreateGrabLine then CreateGrabLine:FireServer(tRoot, Vector3.zero, tRoot.Position, false) end

                        local startTime = tick()
                        while tick() - startTime < 0.7 do
                            if SNO and tRoot.Parent then
                                SNO:FireServer(tRoot, tRoot.CFrame)
                            end
                            Heartbeat:Wait()
                        end

                        blobRoot.CFrame = SavedPos
                        blobRoot.Velocity = Vector3.zero

                        startTime = tick()
                        while tick() - startTime < 0.7 do
                            if SNO and tRoot.Parent then
                                SNO:FireServer(tRoot, tRoot.CFrame)
                            end
                            Heartbeat:Wait()
                        end
                    end
                end
            end

            TeleportToTarget()

            while kickLoopEnabled do
                if not selectedTarget or not selectedTarget.Parent or not selectedTarget.Character then
                    break
                end

                local tChar = selectedTarget.Character
                local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
                local tHum = tChar and tChar:FindFirstChild("Humanoid")

                if not tHum or not tRoot then
                    Heartbeat:Wait()
                    continue
                end

                if tHum.Health > 0 and wasDead then
                    wasDead = false
                    TeleportToTarget()
                    Heartbeat:Wait()
                    continue
                end

                if tHum.Health <= 0 then
                    wasDead = true
                    Heartbeat:Wait()
                    continue
                end

                blobRoot.CFrame = SavedPos
                blobRoot.Velocity = Vector3.zero

                tRoot.CFrame = lockPos
                tRoot.Velocity = Vector3.zero
                tRoot.RotVelocity = Vector3.zero

                if tRoot.AssemblyLinearVelocity then
                    tRoot.AssemblyLinearVelocity = Vector3.zero
                    tRoot.AssemblyAngularVelocity = Vector3.zero
                end

                if SNO then SNO:FireServer(tRoot, lockPos) end

                packetCount = packetCount + 1
                if packetCount >= 2 then
                    packetCount = 0

                    tHum.PlatformStand = true
                    tHum.Sit = true

                    if R_Det then
                        local weld = R_Det:FindFirstChild("RightWeld") or R_Det:FindFirstChildWhichIsA("Weld")
                        if weld and CD then CD:FireServer(weld) end
                    end

                    if DestroyGrabLine then DestroyGrabLine:FireServer(tRoot) end
                    if R_Det and CG then CG:FireServer(R_Det, tRoot, R_Weld) end
                    if CreateGrabLine then CreateGrabLine:FireServer(tRoot, Vector3.zero, tRoot.Position, false) end
                end

                Heartbeat:Wait()
            end

            -- ループ終了
            kickLoopEnabled = false
            if blobRoot then
                blobRoot.CFrame = SavedPos
                blobRoot.Velocity = Vector3.zero
            end
            -- UI更新（安全のため）
            pcall(function()
                toggleButton.Text = "Loop Kick: OFF"
                toggleButton.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
                statusLabel.Text = "Kick loop stopped"
            end)
            loopThread = nil
        end)
    end)

    blobKickUI = screenGui
end

-- ★★★ フラットな丸型配置（半径22・高さ32） ★★★
local function placePlayersInFlatCircle(players, centerPos, radius, height)
    local count = #players
    if count == 0 then return end

    if count == 1 then
        local p = players[1]
        if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = p.Character.HumanoidRootPart
            local pos = Vector3.new(centerPos.X, height, centerPos.Z)
            hrp.CFrame = CFrame.lookAt(pos, Vector3.new(centerPos.X, height, centerPos.Z))
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
        return
    end

    for i, p in ipairs(players) do
        if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = p.Character.HumanoidRootPart
            local theta = ((i - 1) / count) * 2 * math.pi
            local x = radius * math.cos(theta)
            local z = radius * math.sin(theta)
            local pos = Vector3.new(centerPos.X + x, height, centerPos.Z + z)
            hrp.CFrame = CFrame.lookAt(pos, Vector3.new(centerPos.X, height, centerPos.Z))
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end
end

        end

        local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if not humanoid then
            Library:Notify("⚠️ Error", "Humanoid not found", 2)
            return false
        end

        local originalWalkSpeed = humanoid.WalkSpeed
        local originalPosition = LocalPlayer.Character.HumanoidRootPart.CFrame
        humanoid.WalkSpeed = 0

        if not megaMenuToys then
            humanoid.WalkSpeed = originalWalkSpeed
            Library:Notify("⚠️ Error", "MenuToys not found", 2)
            return false
        end

        megaSpawnToyRF:InvokeServer("InstrumentWoodwindOcarina", 
            CFrame.new(184.148834, -5.54824972, 498.136749, 0.829037189, -0.214714944, 0.516328275, 0, 0.923344612, 0.383972496, -0.559193552, -0.318327487, 0.765486956), 
            Vector3.new(0, 34, 0))
        task.wait(0.4)

        local toyFolder = Workspace:FindFirstChild(playerName .. "SpawnedInToys")
        if not toyFolder or not toyFolder:FindFirstChild("InstrumentWoodwindOcarina") then
            humanoid.WalkSpeed = originalWalkSpeed
            Library:Notify("⚠️ Error", "Ocarina not found", 2)
            return false
        end

        local ocarina = toyFolder:FindFirstChild("InstrumentWoodwindOcarina")
        if not ocarina or not ocarina:FindFirstChild("HoldPart") then
            humanoid.WalkSpeed = originalWalkSpeed
            Library:Notify("⚠️ Error", "Ocarina HoldPart not found", 2)
            return false
        end

        ocarina.HoldPart.HoldItemRemoteFunction:InvokeServer(ocarina, Workspace:FindFirstChild(playerName))

        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(304.06, 25.77, 488.54)
        end

        task.wait(0.21)

        if toyFolder and toyFolder:FindFirstChild("InstrumentWoodwindOcarina") then
            megaDestroyToy:FireServer(toyFolder.InstrumentWoodwindOcarina)
        end

        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            LocalPlayer.Character.HumanoidRootPart.CFrame = originalPosition
        end

        task.wait(0.7)

        megaSpawnToyRF:InvokeServer("Campfire", 
            CFrame.new(257.638672, -5.57392979, 450.103638, -0.950906992, -0.171067372, 0.257899135, 0, 0.833338678, 0.552762806, -0.309477001, 0.525626004, -0.79242748), 
            Vector3.new(0, 161.9720001220703, 0))

        task.wait(0.7)

        local campfirePosition = Vector3.new(257.638672, -5.57392979, 450.103638)
        local toyFolder2 = Workspace:FindFirstChild(playerName .. "SpawnedInToys")
        if toyFolder2 and toyFolder2:FindFirstChild("Campfire") then
            local campfire = toyFolder2:FindFirstChild("Campfire")
            local primaryPart = campfire.PrimaryPart or campfire:FindFirstChildWhichIsA("BasePart")
            if primaryPart then
                local distance = (primaryPart.Position - campfirePosition).Magnitude
                if distance < 10 then
                    task.wait(1)
                    humanoid.WalkSpeed = originalWalkSpeed
                    Library:Notify("✅ Success", "Barrier破壊に成功！", 2)
                    return true
                end
            end
        end

        humanoid.WalkSpeed = originalWalkSpeed
        Library:Notify("⚠️ Failed", "Campfire placement failed", 2)
        return false
    end)

    if not success then
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if humanoid then humanoid.WalkSpeed = 16 end
        end
        Library:Notify("❌ Error", "Execution error: " .. tostring(result), 2)
        return false
    end

    autoBarrierRunning = false
    return result
end

    if plotItems then
        local playersInPlots = plotItems:FindFirstChild("PlayersInPlots")

            return false
        end
    end

            error("キャラクターが見つかりません") 
        end
        
        local humanoid = player.Character:FindFirstChildOfClass("Humanoid")

            error("Humanoidが見つかりません")

            humanoid.WalkSpeed = 16 
            error("MenuToysが見つかりません")

            CFrame.new(184.148834, -5.54824972, 498.136749, 0.829037189, -0.214714944, 0.516328275, 0, 0.923344612, 0.383972496, -0.559193552, -0.318327487, 0.765486956), 
            Vector3.new(0, 34, 0))
        
        if not spawnResult then 
            humanoid.WalkSpeed = 16 
            error("オカリナのスポーンに失敗しました") 
        end
        task.wait(0.4)
        
        local toyFolder = Workspace:FindFirstChild(playerName .. "SpawnedInToys")
        if not toyFolder then 
            humanoid.WalkSpeed = 16 
            error("トイフォルダが見つかりません") 
        end
        
        local ocarina = toyFolder:FindFirstChild("InstrumentWoodwindOcarina")

        end

            humanoid.WalkSpeed = 16 
            error("オカリナが既に消失しています") 
        end
        
        megaDestroyToy:FireServer(Workspace[playerName .. "SpawnedInToys"].InstrumentWoodwindOcarina)
        task.wait(1.5)
        
        local campfireSpawnResult = megaSpawnToyRF:InvokeServer("Campfire",

            Vector3.new(0, 161.9720001220703, 0))

            humanoid.WalkSpeed = 16 
            error("Campfireのスポーンに失敗しました") 
        end
        task.wait(1)
        
        if not Workspace:FindFirstChild(playerName .. "SpawnedInToys") then 
            humanoid.WalkSpeed = 16 
            error("トイフォルダが消失しました") 
        end

            error("Campfireが見つかりません")

        return true
    end)

        if humanoid then humanoid.WalkSpeed = 16 end
        Library:Notify("❌ 失敗", tostring(err), 2)
        return false
