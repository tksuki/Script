        if not AKV2_Shuriken or not AKV2_Shuriken.Parent then

            AKV2_Shuriken = ANV2_spawnToy("NinjaShuriken", targetPart.CFrame + Vector3.new(0, 15, 0))
            if not AKV2_Shuriken then return false end

        if not sp then AKV2_destroyShuriken() return false end

        local spe = ge and ge:FindFirstChild("SetNetworkOwner")
        if spe then pcall(function() spe:FireServer(sp, targetPart, CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(90), 0, 0)) end) end

        while true do
            task.wait()
            if sw.Part1 == targetPart then break end
            if not sw.Parent or not targetPart.Parent or not sp.Parent or not AKV2_Shuriken or not AKV2_Shuriken.Parent then break end

        end

                    local t0 = tick()
                    repeat
                        RunService.Heartbeat:Wait()
                        tf = Workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
                        toy = tf and tf:FindFirstChild(AntiInputLagToy)
                    until toy or tick() - t0 > 1
                end
                if toy and toy.Parent then

                    if hp then
                        local holdingPlayer = hp:FindFirstChild("HoldingPlayer")
                        holdingPlayer = holdingPlayer and holdingPlayer.Value
                        if holdingPlayer and holdingPlayer ~= LocalPlayer then

                                RunService.Heartbeat:Wait()

                            end)
                        end
                    end

            end
        end)

    local function breakVehicleSeat(seat, mh, mhr)
        if not seat or not seat.Parent then return false end
        if not (seat:IsA("VehicleSeat") or seat:IsA("Seat")) then return false end
        if seat.Occupant or seat:GetAttribute("broken") then return false end
        if not mh or mh.Health == 0 or not mhr then return false end
        mh.Sit = false
        local broke = false
        local conn = seat.ChildAdded:Connect(function(child)
            if child.Name == "SeatWeld" then pcall(function() child.Enabled = false end) child:Destroy() broke = true end
        end)
        local t0 = tick()
        while seat.Parent and mh.Parent and not broke and tick() - t0 <= 0.5 do
            RunService.Heartbeat:Wait()
            pcall(function() seat:Sit(mh) end)
        end
        mh.Sit = false
        if conn then conn:Disconnect() end
        if broke then seat:SetAttribute("broken", true) return true end

    local function getOptionValue(name, fallback)
        local opt = EB.Options[name]
        if not opt then return fallback end
        local ok, val = pcall(function() return opt.Value end)
        if ok and val ~= nil then return val end
        ok, val = pcall(function() return opt:GetValue() end)
        if ok and val ~= nil then return val end
        return fallback
    end
    local function getSelectedSongName() return getOptionValue("SongSelect", nil) end
    local function getSeekStart() return getOptionValue("SeekStart", 0) end
    local function isLoopEnabled() return getOptionValue("LoopMusic", true) == true end
    local function isPlayOn() return getOptionValue("PlayMusic", false) == true end

    local function applyDistortion()
        local sound = getSound()
        if not sound then return end
        local d = sound:FindFirstChild("Distortion")
        if not d then return end
        local on = getOptionValue("DistortionToggle", false) == true
        local level = tonumber(getOptionValue("DistortionLevel", 50)) or 50
        d.Enabled = on
        d.Level = math.clamp(level / 1000, 0, 1)
    end
    local function applyBoost()
        local sound = getSound()
        if not sound then return end
        local b = sound:FindFirstChild("Boost")
        if not b then return end
        local on = getOptionValue("BoostToggle", false) == true
        local level = tonumber(getOptionValue("BoostLevel", 50)) or 50
        b.Enabled = on
        local gain = (level / 1000) * 30
        b.LowGain = gain
        b.MidGain = gain
        b.HighGain = gain
    end
    local function playCurrent()
        local sound = getSound()
        if not sound then return end
        local selectedName = getSelectedSongName()
        local selectedSong = selectedName and Songs[selectedName] or nil
        if not selectedSong or selectedSong.AssetAudioPath == "" then
            v3:Notify({ Title = "Emukunhub Free", Description = "This song could not be loaded.", Time = 3 })
            return
        end
        sound:Stop()
        sound.SoundId = selectedSong.AssetAudioPath
        sound.TimePosition = math.max(0, getSeekStart())
        sound:Play()
        applyDistortion()
        applyBoost()
    end

-- ============================================================
--  Server tab — Break (Plot Break + Plot Break v2 + Ground Kick)
-- ============================================================
do
    local LocalPlayer       = EB.LocalPlayer
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local Workspace         = game:GetService("Workspace")

    local function findRemote(name, className)
        local direct = ReplicatedStorage:FindFirstChild(name)
        if direct and direct:IsA(className) then return direct end
        for _, container in ipairs(ReplicatedStorage:GetChildren()) do
            if container:IsA("Folder") or container:IsA("Model") then
                local found = container:FindFirstChild(name)
                if found and found:IsA(className) then return found end
            end
        end
        local waited = ReplicatedStorage:WaitForChild(name, 3)
        if waited and waited:IsA(className) then return waited end
        return nil
    end

    local SetNetworkOwner        = findRemote("SetNetworkOwner", "RemoteEvent")
    local DestroyToyRemote       = findRemote("DestroyToy", "RemoteEvent")
    local SpawnToyRemoteFunction = findRemote("SpawnToyRemoteFunction", "RemoteFunction")

    local playerEvents = ReplicatedStorage:FindFirstChild("PlayerEvents")
    local stickyEvent  = playerEvents and playerEvents:FindFirstChild("StickyPartEvent")

    local function getHRP(plr)
        local char = plr.Character or plr.CharacterAdded:Wait()
        return char:WaitForChild("HumanoidRootPart", 5)
    end

    local function getTargetContainer(plr)
        local plotItems = Workspace:FindFirstChild("PlotItems")
        local plots     = Workspace:FindFirstChild("Plots")
        if plotItems and plots then
            local inPlots = plotItems:FindFirstChild("PlayersInPlots")
            if inPlots and inPlots:FindFirstChild(plr.Name) then
                for _, plot in ipairs(plots:GetChildren()) do
                    local sign   = plot:FindFirstChild("PlotSign")
                    local owners = sign and sign:FindFirstChild("ThisPlotsOwners")
                    if owners then
                        for _, owner in ipairs(owners:GetChildren()) do
                            if owner.Value == plr.Name then
                                local folder = plotItems:FindFirstChild(plot.Name)
                                if folder then return folder end
                            end
                        end
                    end
                end
            end
        end
        return Workspace:FindFirstChild(plr.Name .. "SpawnedInToys")
    end

    local function destroyShuriken(sh)
        if sh and sh.Parent and DestroyToyRemote then
            pcall(function() DestroyToyRemote:FireServer(sh) end)
        end
    end

    local function spawnAndAttachShuriken(targetPart, container, attachCFrame)
        if not targetPart or not container then return nil end
        local plr = LocalPlayer
        local canSpawn = plr:FindFirstChild("CanSpawnToy")
        if not canSpawn then return nil end
        local hrp = getHRP(plr)
        if not hrp then return nil end
        local preExisting = {}
        for _, child in ipairs(container:GetChildren()) do
            if child.Name == "NinjaShuriken" then table.insert(preExisting, child) end
        end
        if not canSpawn.Value or not SpawnToyRemoteFunction then return nil end
        local newShuriken = nil
        pcall(function()
            SpawnToyRemoteFunction:InvokeServer("NinjaShuriken", hrp.CFrame * CFrame.new(0, 2, 8), Vector3.new(0, 0, 0))
        end)
        local startTime = tick()
        while tick() - startTime < 2 do
            for _, child in ipairs(container:GetChildren()) do
                if child.Name == "NinjaShuriken" and not table.find(preExisting, child) then
                    newShuriken = child
                    break
                end
            end
            if newShuriken then break end
            task.wait()
        end
        if not newShuriken then return nil end
        local soundPart = newShuriken:FindFirstChild("SoundPart") or newShuriken:WaitForChild("SoundPart", 0.1)
        if soundPart and SetNetworkOwner then
            pcall(function() SetNetworkOwner:FireServer(soundPart, soundPart.CFrame) end)
        end
        local sticky = newShuriken:FindFirstChild("StickyPart")
        if sticky and stickyEvent then
            pcall(function()
                stickyEvent:FireServer(sticky, targetPart, attachCFrame or CFrame.new(
                    9.99999996e+11, 9.99999996e+11, 9.99999996e+11,
                    1, 0, 0, 0, 1, 0, 0, 0, 1
                ))
            end)
        end
        return newShuriken
    end

    local BreakGroup = Tabs.Server:AddLeftGroupbox("Break", "hammer")

    local plotSelected      = "Plot1"
    local plotMainActive    = false
    local plotMainThread    = nil
    local plotMainShurikens = {}

    local function plotCleanupList(list)
        for _, sh in ipairs(list) do destroyShuriken(sh) end
        for i = #list, 1, -1 do list[i] = nil end
    end

    local function plotSpawnAndAttach(plotArea)
        if not plotArea then return nil end
        local container = getTargetContainer(LocalPlayer)
        if not container then return nil end
        return spawnAndAttachShuriken(plotArea, container, CFrame.new(999999999600, 999999999600, 999999999600))
    end

    local function plotMainLoop()
        while plotMainActive do
            local plots = Workspace:FindFirstChild("Plots")
            local plot  = plots and plots:FindFirstChild(plotSelected)
            if not plot then
                v3:Notify({Title = "Plot Break", Description = "Plot not found: " .. tostring(plotSelected), Time = 2})
                plotMainActive = false
                break
            end
            local plotArea = plot:FindFirstChild("PlotArea")
            if not plotArea then
                v3:Notify({Title = "Plot Break", Description = "PlotArea not found", Time = 2})
                plotMainActive = false
                break
            end
            local already = false
            for _, sh in ipairs(plotMainShurikens) do
                local sp = sh and sh:FindFirstChild("StickyPart")
                local sw = sp and sp:FindFirstChild("StickyWeld")
                if sw and sw.Part1 == plotArea then already = true break end
            end
            if not already then
                local sh = plotSpawnAndAttach(plotArea)
                if sh then table.insert(plotMainShurikens, sh) end
            end
            local alive = {}
            for _, sh in ipairs(plotMainShurikens) do
                if sh and sh.Parent then table.insert(alive, sh) end
            end
            plotMainShurikens = alive
            task.wait(1)
        end
        plotCleanupList(plotMainShurikens)
    end

    BreakGroup:AddDropdown("PlotBreakSelect", {
        Text = "Target Plot",
        Values = { "Plot1", "Plot2", "Plot3", "Plot4", "Plot5" },
        Default = "Plot1",
        Multi = false,
        Callback = function(Value)
            if type(Value) == "table" then plotSelected = Value[1]
            else plotSelected = Value end
        end,
    })

    BreakGroup:AddToggle("PlotBreakMainToggle", {
        Text = "Plot Break", Default = false,
        Tooltip = "Break selected plot only / OFF = stop + remove all shurikens",
        Callback = function(Value)
            if Value then
                if plotMainActive then return end
                plotMainActive = true
                plotMainThread = task.spawn(plotMainLoop)
                v3:Notify({Title = "Plot Break", Description = "Started (" .. tostring(plotSelected) .. ")", Time = 2})
            else
                plotMainActive = false
                if plotMainThread then pcall(function() task.cancel(plotMainThread) end) plotMainThread = nil end
                local count = #plotMainShurikens
                plotCleanupList(plotMainShurikens)
                v3:Notify({Title = "Plot Break", Description = string.format("Stopped (%d removed)", count), Time = 2})
            end
        end,
    })

    local plotV2Active    = false
    local plotV2Thread    = nil
    local plotV2Shurikens = {}

    local function plotV2Loop()
        while plotV2Active do
            local plots = Workspace:FindFirstChild("Plots")
            if not plots then task.wait(1) continue end
            for i = 1, 5 do
                if not plotV2Active then break end
                local plot     = plots:FindFirstChild("Plot" .. i)
                local plotArea = plot and plot:FindFirstChild("PlotArea")
                if plotArea then
                    local already = false
                    for _, sh in ipairs(plotV2Shurikens) do
                        local sp = sh and sh:FindFirstChild("StickyPart")
                        local sw = sp and sp:FindFirstChild("StickyWeld")
                        if sw and sw.Part1 == plotArea then already = true break end
                    end
                    if not already then
                        local sh = plotSpawnAndAttach(plotArea)
                        if sh then table.insert(plotV2Shurikens, sh) end
                        task.wait(0.05)
                    end
                end
            end
            local alive = {}
            for _, sh in ipairs(plotV2Shurikens) do
                if sh and sh.Parent then table.insert(alive, sh) end
            end
            plotV2Shurikens = alive
            task.wait(0.5)
        end
        plotCleanupList(plotV2Shurikens)
    end

    BreakGroup:AddToggle("PlotBreakV2MainToggle", {
        Text = "Plot Break v2", Default = false,
        Tooltip = "Break all 5 plots at once / OFF = stop + remove all shurikens",
        Callback = function(Value)
            if Value then
                if plotV2Active then return end
                plotV2Active = true
                plotV2Thread = task.spawn(plotV2Loop)
                v3:Notify({Title = "Plot Break v2", Description = "Started (all 5 plots)", Time = 2})
            else
                plotV2Active = false
                if plotV2Thread then pcall(function() task.cancel(plotV2Thread) end) plotV2Thread = nil end
                local count = #plotV2Shurikens
                plotCleanupList(plotV2Shurikens)
                v3:Notify({Title = "Plot Break v2", Description = string.format("Stopped (%d removed)", count), Time = 2})
            end
        end,
    })

    local groundKickEnabled = false
    local groundKickThread  = nil
    local groundKickToys    = {}

    local function cleanupGroundKickToys()
        for _, toy in ipairs(groundKickToys) do destroyShuriken(toy) end
        groundKickToys = {}
    end

    local function groundKickBreakOnce()
        local plr = LocalPlayer
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local rayParams = RaycastParams.new()
        rayParams.FilterDescendantsInstances = { char }
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        local result = Workspace:Raycast(hrp.Position, Vector3.new(0, -10, 0), rayParams)
        local hit = result and result.Instance
        if not hit or not hit:IsA("BasePart") or hit:IsA("Terrain") then return end
        local container = getTargetContainer(plr)
        if not container then return end
        local newToy = spawnAndAttachShuriken(hit, container)
        if newToy then table.insert(groundKickToys, newToy) end
    end

    local function StartGroundKick()
        if groundKickEnabled then return end
        groundKickEnabled = true
        groundKickThread = task.spawn(function()
            while groundKickEnabled do
                groundKickBreakOnce()
                task.wait(0.8)
            end
            cleanupGroundKickToys()
        end)
        v3:Notify({Title = "Ground Kick", Description = "ON", Time = 2})
    end

    local function StopGroundKick()
        groundKickEnabled = false
        if groundKickThread then pcall(function() task.cancel(groundKickThread) end) groundKickThread = nil end
        cleanupGroundKickToys()
        v3:Notify({Title = "Ground Kick", Description = "OFF", Time = 2})
    end

    BreakGroup:AddToggle("GroundKickToggle", {
        Text = "Ground Kick", Default = false,
        Tooltip = "Auto-break the ground beneath you using NinjaShuriken",
        Callback = function(Value)
            if Value then StartGroundKick() else StopGroundKick() end
        end,
    })

    BreakGroup:AddButton({
        Text = "Stop All Break",
        Tooltip = "Stop Plot Break / Plot Break v2 / Ground Kick and remove all shurikens",
        Func = function()
            plotMainActive = false
            if plotMainThread then pcall(function() task.cancel(plotMainThread) end) plotMainThread = nil end
            plotCleanupList(plotMainShurikens)
            plotV2Active = false
            if plotV2Thread then pcall(function() task.cancel(plotV2Thread) end) plotV2Thread = nil end
            plotCleanupList(plotV2Shurikens)
            StopGroundKick()
            v3:Notify({Title = "Break", Description = "All stopped", Time = 2})
        end,
    })

    if v3 and v3.OnUnload then
        v3:OnUnload(function()
            plotMainActive = false
            if plotMainThread then pcall(function() task.cancel(plotMainThread) end) plotMainThread = nil end
            plotCleanupList(plotMainShurikens)
            plotV2Active = false
            if plotV2Thread then pcall(function() task.cancel(plotV2Thread) end) plotV2Thread = nil end
            plotCleanupList(plotV2Shurikens)
            StopGroundKick()
        end)
    end
end
