local antiRagdollConnections = {}
local antiRagdollEnabled = false

local function clearAntiRagdollConnections()
    for _, conn in ipairs(antiRagdollConnections) do
        if conn and conn.Disconnect then pcall(function() conn:Disconnect() end) end
    end
    antiRagdollConnections = {}
end

local function protectHumanoid(humanoid)
    if not humanoid or not antiRagdollEnabled then return end
    pcall(function()
        humanoid.BreakJointsOnDeath = false
        humanoid.AutoRotate = true
        humanoid.PlatformStand = false
    end)
    table.insert(antiRagdollConnections, humanoid.HealthChanged:Connect(function(health)
        if antiRagdollEnabled and health <= 0 and humanoid.Parent then
            pcall(function() humanoid.Health = 1 end)
        end
    end))
    table.insert(antiRagdollConnections, humanoid:GetPropertyChangedSignal("AutoRotate"):Connect(function()
        if antiRagdollEnabled and humanoid.AutoRotate == false and humanoid.Parent then
            pcall(function() humanoid.AutoRotate = true end)
        end
    end))
    table.insert(antiRagdollConnections, humanoid:GetPropertyChangedSignal("PlatformStand"):Connect(function()
        if antiRagdollEnabled and humanoid.PlatformStand == true and humanoid.Parent then
            pcall(function() humanoid.PlatformStand = false end)
        end
    end))
    table.insert(antiRagdollConnections, RunService.RenderStepped:Connect(function()
        if antiRagdollEnabled and humanoid and humanoid.Parent then
            if humanoid.Sit and humanoid.SeatPart == nil then
                pcall(function() humanoid.Sit = false end)
            end
        end
    end))
end

local function onCharacterAddedForProtection(character)
    local humanoid = character:FindFirstChild("Humanoid")
    if humanoid and antiRagdollEnabled then protectHumanoid(humanoid) end
end

local function setAntiRagdollProtection(enabled)
    antiRagdollEnabled = enabled
    if enabled then
        LocalPlayer.CharacterAdded:Connect(onCharacterAddedForProtection)
        local character = LocalPlayer.Character
        local humanoid = character and character:FindFirstChild("Humanoid")
        if humanoid then protectHumanoid(humanoid) end
    else
        clearAntiRagdollConnections()
    end
end

local function GetRepresentativePart(character, partKey)
    if not character then return nil end
    local function findAny(names)
        for _, n in ipairs(names) do
            local p = character:FindFirstChild(n)
            if p and p:IsA("BasePart") then return p end
        end
    end
    if partKey=="Head" then return findAny({"Head"})
    elseif partKey=="HumanoidRootPart" then return findAny({"HumanoidRootPart","Torso","UpperTorso","LowerTorso"})
    elseif partKey=="Left Arm" then return findAny({"LeftHand","LeftUpperArm","Left Arm","LeftLowerArm"})
    elseif partKey=="Right Arm" then return findAny({"RightHand","RightUpperArm","Right Arm","RightLowerArm"})
    elseif partKey=="Left Leg" then return findAny({"LeftFoot","LeftLowerLeg","Left Leg","LeftUpperLeg"})
    elseif partKey=="Right Leg" then return findAny({"RightFoot","RightLowerLeg","Right Leg","RightUpperLeg"})
    end
end

local function GetPart(character, partKey)
    if not character then return {} end
    local names = {}
    if partKey=="HRP" then
        names={"HumanoidRootPart","Torso","UpperTorso"}
    elseif partKey=="Others" then
        names={"Head","Left Arm","Right Arm","Left Leg","Right Leg",
            "LeftUpperArm","RightUpperArm","LeftLowerArm","RightLowerArm",
            "LeftHand","RightHand","LeftUpperLeg","RightUpperLeg",
            "LeftLowerLeg","RightLowerLeg","LeftFoot","RightFoot"}
    end
    local found={}
    for _, name in ipairs(names) do
        local p=character:FindFirstChild(name)
        if p and p:IsA("BasePart") then table.insert(found,p) end
    end
    return found
end

local function GetPlayerList()
    local list={}
    for _, p in pairs(Players:GetPlayers()) do
        if p~=LocalPlayer then table.insert(list,p.Name) end
    end
    return list
end

local function GetPlayerListDisplay()
    local list={}
    for _, p in ipairs(Players:GetPlayers()) do
        if p~=LocalPlayer then table.insert(list, p.DisplayName.." @"..p.Name) end
    end
    if #list==0 then table.insert(list,"No Players") end
    return list
end

local function ExtractUsername(displayValue)
    if not displayValue or displayValue=="" or displayValue=="No Players" then return nil end
    return displayValue:match("@(.+)$")
end

local function isWhitelisted(player)
    if not CFG.whitelistEnabled then return false end
    if player==LocalPlayer then return true end
    if LocalPlayer:IsFriendsWith(player.UserId) then return true end
    if State.whitelistedPlayers[player.Name] then return true end
    return false
end

local function addToWhitelist(playerName)
    State.whitelistedPlayers[playerName]=true
    Library:Notify("Whitelist: Added "..playerName, 3)
end

local function getWhitelistInfo()
    local info="[Protected Players]\n"
    local count=0
    info=info.."\n[Friends]\n"
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr~=LocalPlayer and LocalPlayer:IsFriendsWith(plr.UserId) then
            info=info.."- "..plr.Name.."\n"; count=count+1
        end
    end
    local mc=0
    for name, _ in pairs(State.whitelistedPlayers) do
        if mc==0 then info=info.."\n[Manually Added]\n" end
        info=info.."- "..name.."\n"; mc=mc+1
    end
    if count+mc==0 then return "No protected players" end
    return info
end

    pcall(function()

end
local function handleSpinGrab(model)
    pcall(function()
        local gp=model:FindFirstChild("GrabPart")
        local wc=gp and gp:FindFirstChild("WeldConstraint")
        if wc and wc.Part1 then

            while enabled() do

local function StartRagdollSpam()

                if hrp then ragdollRemote:FireServer(hrp,1) end
            end)
            task.wait(0.05)
        end
    end)
end
local function StopRagdollSpam() CFG.ragdollRunning=false end

local function UpdateRagdollState()
    if CFG.headEnabled or CFG.legsEnabled or CFG.obeyPartEnabled or CFG.obeyAllEnabled or CFG.feetTPDeleteEnabled or CFG.danceEnabled then
        StartRagdollSpam()
    else StopRagdollSpam() end
end

local function executeSequenceGucci(child)
    if child.Name~=CFG.targetToyName then return end
    CFG.currentInstance=child
    local _, hrp, hum=getCharInfo()
    if not (hrp and hum) then return end
    local seat=child:WaitForChild("VehicleSeat",2) or child:FindFirstChildWhichIsA("VehicleSeat",true)
    if seat and hum then
        pcall(function() seat:Sit(hum) end)
        local start=tick()
        while tick()-start<CONST.ANTI_DURATION and CFG.autoRunEnabled do
            pcall(function() ragdollRemote:FireServer(hrp,0); hum:ChangeState(Enum.HumanoidStateType.Jumping) end)
            RunService.Heartbeat:Wait()
        end
        local primary=child.PrimaryPart or child:FindFirstChild("HumanoidRootPart",true) or child:FindFirstChild("Part",true)
        if primary and CFG.autoRunEnabled then
            pcall(function()
                if primary.SetNetworkOwner then primary:SetNetworkOwner(LocalPlayer) end
                sethiddenproperty(primary,"NetworkIsSleeping",false)
            end)
        end
        if CFG.originalPos then hrp.CFrame=CFG.originalPos end
        task.wait(0.1)
        if seat then seat:Destroy() end
    end
end

local function startSystem()
    local _, hrp=getCharInfo()
    if hrp then CFG.originalPos=hrp.CFrame end
    if CFG.activeConnection then CFG.activeConnection:Disconnect() end
    CFG.activeConnection=toysFolder.ChildAdded:Connect(function(child)
        task.spawn(function() executeSequenceGucci(child) end)
    end)
    task.spawn(function()
        while CFG.autoRunEnabled do
            if not CFG.currentInstance or not CFG.currentInstance.Parent then
                pcall(function() SpawnToyRemoteFunction:InvokeServer(CFG.targetToyName,CFrame.new(CONST.SPAWN_POS),Vector3.new(0,-15.716,0)) end)
            end
            task.wait(1)
        end
    end)
end

local function runAura(withHP, runningFlagGetter)
    local hpToggle=false
    while runningFlagGetter() do
        task.wait(0.01)
        local char, hrp=getCharInfo(); if not hrp then continue end
        local blobman=spawnBlobmanIfNeeded(); if not blobman then continue end
        sitOnBlobman(blobman)
        local sf=blobman:FindFirstChild("BlobmanSeatAndOwnerScript")
        local det=blobman:FindFirstChild("LeftDetector")
        local weld=det and det:FindFirstChild("LeftWeld")
        if not (sf and det and weld) then continue end
        local Grab=sf:FindFirstChild("CreatureGrab"); local Release=sf:FindFirstChild("CreatureRelease"); local Drop=sf:FindFirstChild("CreatureDrop")
        if not (Grab and Release and Drop) then continue end
        hpToggle=not hpToggle
        for _, plr in ipairs(Players:GetPlayers()) do
            if not runningFlagGetter() then break end
            if plr~=LocalPlayer and not isWhitelisted(plr) then
                local c=plr.Character; local thrp=c and c:FindFirstChild("HumanoidRootPart"); local thum=c and c:FindFirstChildOfClass("Humanoid")

local function runTargetKillAura(targetPlayer)
    while CFG.targetKillRunning and CFG.selectedTargetPlayer==targetPlayer do
        task.wait(0.01)
        local char, hrp=getCharInfo(); if not hrp then continue end
        local target=Players:FindFirstChild(targetPlayer)
        if not target or not target.Character then task.wait(1); continue end
        local tChar=target.Character; local tHRP=tChar:FindFirstChild("HumanoidRootPart"); local tHum=tChar:FindFirstChildOfClass("Humanoid")
        if not (tHRP and tHum) or tHum.Health<=0 then task.wait(1); continue end
        local blobman=spawnBlobmanIfNeeded(); if not blobman then task.wait(0.5); continue end
        local seat=blobman:FindFirstChildWhichIsA("VehicleSeat",true)
        local hum=char and char:FindFirstChildOfClass("Humanoid")
        if not (seat and hum) then task.wait(0.5); continue end
        if not seat.Occupant then pcall(function() seat:Sit(hum) end); task.wait(0.2) end
        if not seat.Occupant then task.wait(0.5); continue end
        local sf=blobman:FindFirstChild("BlobmanSeatAndOwnerScript")
        local det=blobman:FindFirstChild("LeftDetector"); local weld=det and det:FindFirstChild("LeftWeld")
        if not (sf and det and weld) then task.wait(0.5); continue end
        local Grab=sf:FindFirstChild("CreatureGrab"); local Release=sf:FindFirstChild("CreatureRelease"); local Drop=sf:FindFirstChild("CreatureDrop")
        if not (Grab and Release and Drop) then task.wait(0.5); continue end
        CFG.lastTargetPosition=hrp.CFrame
        pcall(function() hrp.CFrame=tHRP.CFrame*CFrame.new(0,0,-3) end)
        task.wait(0.2)
        local st=tick()
        while tick()-st<2 and CFG.targetKillRunning do
            if not (target and target.Character and tHum.Health>0) then break end
            if not seat.Occupant then break end
            if (hrp.Position-tHRP.Position).Magnitude<=30 then
                pcall(function()
                    Grab:FireServer(det,tHRP,weld); Release:FireServer(weld,tHRP); Drop:FireServer(weld,tHRP)
                    tHum.Health=0
                end)
            end
            task.wait(0.01)
        end
        task.wait(0.1)
        if CFG.lastTargetPosition then pcall(function() hrp.CFrame=CFG.lastTargetPosition end) end
        if tHum.Health<=0 then
            local respawned=false
            local conn2; conn2=targetPlayer and Players:FindFirstChild(targetPlayer) and Players:FindFirstChild(targetPlayer).CharacterAdded:Connect(function() respawned=true; if conn2 then conn2:Disconnect() end end)
            local waitStart=tick()
            while not respawned and CFG.targetKillRunning and tick()-waitStart<30 do task.wait() end
            if conn2 then pcall(function() conn2:Disconnect() end) end
            task.wait(0.05)
        end
    end
end

local function SetupPhysics(Part)
    local BP=Part:FindFirstChildOfClass("BodyPosition") or Instance.new("BodyPosition")
    local BG=Part:FindFirstChildOfClass("BodyGyro") or Instance.new("BodyGyro")
    BP.P=15000; BP.D=200; BP.MaxForce=Vector3.new(1,1,1)*1e10; BP.Parent=Part
    BG.P=15000; BG.D=200; BG.MaxTorque=Vector3.new(1,1,1)*1e10; BG.Parent=Part
    return BG, BP
end

local function BuildWings()
    CleanupWings()
    local folders=GetToyFolders()
    local allSpawnedToys={}
    for _, folder in ipairs(folders) do
        for _, x in ipairs(folder:GetDescendants()) do
            if x:IsA("Model") and x.Name==CFG.SelectedItem then table.insert(allSpawnedToys,x) end
        end
    end
    if #allSpawnedToys==0 then return false end
    for i=1,2 do
        local Segments={}
        for x=1,CFG.WingLength do
            local p=Instance.new("Part")
            p.CanCollide=false; p.Anchored=true; p.Transparency=1; p.Size=Vector3.new(4,1,4); p.Parent=Workspace
            Segments[#Segments+1]={Part=p}
        end
        local h=Instance.new("Part")
        h.CanCollide=false; h.Anchored=true; h.Transparency=1; h.Size=Vector3.new(4,1,4); h.Parent=Workspace
        State.Wings[#State.Wings+1]={Handle=h,Segments=Segments,Sync={},Reserved=nil}
    end
    for i, v in ipairs(allSpawnedToys) do
        local Side=(i<=(#allSpawnedToys/2)) and 1 or 2
        local Pallet=v:FindFirstChild("SoundPart") or v:FindFirstChild("Handle") or v:FindFirstChildWhichIsA("BasePart")
        if Pallet then
            for _, child in pairs(v:GetChildren()) do
                if child:IsA("BasePart") then child.CanCollide=false end
            end
            local BG, BP=SetupPhysics(Pallet)
            if not State.Wings[Side].Reserved then
                State.Wings[Side].Reserved={BG=BG,BP=BP}
            else
                table.insert(State.Wings[Side].Sync,{BG=BG,BP=BP})
            end
        end
    end
    return true
end

local tpState={isActive=false,savedCF=nil,conns={}}
local tpPerFrame=5

local function tpCoord() local e=10^19; return (math.random()*10)*e*(math.random(1,2)==1 and 1 or -1) end
local function tpDoTP()
    local _, hrp=getCharInfo()
    if hrp then pcall(function() hrp.CFrame=CFrame.new(tpCoord(),tpCoord(),tpCoord()) end) end
end
local function tpStart()
    local _, hrp=getCharInfo(); if hrp then tpState.savedCF=hrp.CFrame end
    tpState.conns[1]=RunService.RenderStepped:Connect(function() for _=1,tpPerFrame do tpDoTP() end end)
    tpState.conns[2]=RunService.Heartbeat:Connect(function() for _=1,tpPerFrame do tpDoTP() end end)
    tpState.conns[3]=RunService.Stepped:Connect(function() for _=1,tpPerFrame do tpDoTP() end end)
    task.spawn(function()
        while tpState.isActive do for _=1,tpPerFrame do tpDoTP() end; task.wait() end
    end)
end
local function tpStop()
    tpState.isActive=false
    for i=1,3 do
        if tpState.conns[i] then tpState.conns[i]:Disconnect(); tpState.conns[i]=nil end
    end
    tpState.conns={}
    if tpState.savedCF then
        local _, hrp=getCharInfo(); if hrp then pcall(function() hrp.CFrame=tpState.savedCF end) end
    end
end

local function LoopKickBlobFunction(targetName)
    local GE=ReplicatedStorage:WaitForChild("GrabEvents")
    local REMOTE_DELAY=0.002; local lastRemote=0
    local char=LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local hum=char:WaitForChild("Humanoid"); local seat=hum.SeatPart
    if not seat or seat.Parent.Name~="CreatureBlobman" then
        Library:Notify("Loop Kick: Please ride Blobman first",3); CFG.loopKickBlobActive=false; return
    end
    local blob=seat.Parent; local blobRoot=blob:FindFirstChild("HumanoidRootPart") or blob.PrimaryPart
    local scriptObj=blob:WaitForChild("BlobmanSeatAndOwnerScript")
    local CG=scriptObj:WaitForChild("CreatureGrab"); local CD=scriptObj:WaitForChild("CreatureDrop")
    local R_Det=blob:WaitForChild("RightDetector")
    local skyBase=CFrame.new(20000,20000,20000)
    if CFG.loopKickBlobMode=="TP to Sky" then
        ensureSkyKickFloor()
        local standPos=CFrame.new(20000,20010.5,20000)
        pcall(function() blobRoot.CFrame=standPos end)
        local myHRP=char:FindFirstChild("HumanoidRootPart")
        if myHRP then pcall(function() myHRP.CFrame=standPos end) end
        task.wait(0.2); skyBase=standPos
    end
    local savedPos=(CFG.loopKickBlobMode=="TP to Sky") and skyBase or blobRoot.CFrame
    local dragging=false; local grabStartTime=0
    while CFG.loopKickBlobActive do
        local target=Players:FindFirstChild(targetName); if not target then break end
        char=LocalPlayer.Character; hum=char and char:FindFirstChild("Humanoid"); seat=hum and hum.SeatPart
        if not seat or seat.Parent.Name~="CreatureBlobman" then break end
        blob=seat.Parent; blobRoot=blob:FindFirstChild("HumanoidRootPart") or blob.PrimaryPart
        local tChar=target.Character; local tRoot=tChar and tChar:FindFirstChild("HumanoidRootPart")
        local tHum=tChar and tChar:FindFirstChild("Humanoid")
        if tRoot and tHum and tHum.Health>0 and blobRoot then
            tRoot.Velocity=Vector3.zero
            if not dragging then
                blobRoot.CFrame=tRoot.CFrame; blobRoot.Velocity=Vector3.zero
                if tick()-lastRemote>=REMOTE_DELAY then
                    lastRemote=tick()
                    pcall(function()
                        tHum.PlatformStand=true; tHum.Sit=true
                        GE.SetNetworkOwner:FireServer(tRoot,blobRoot.CFrame)
                        GE.DestroyGrabLine:FireServer(tRoot)
                    end)
                end
                if grabStartTime==0 then grabStartTime=tick() end
                if tick()-grabStartTime>0.35 then
                    dragging=true; grabStartTime=0
                    blobRoot.CFrame=savedPos; blobRoot.Velocity=Vector3.zero
                end
            else
                blobRoot.CFrame=savedPos; blobRoot.Velocity=Vector3.zero
                local lockPos=savedPos*CFrame.new(0,23,0)
                tRoot.CFrame=lockPos; tHum.PlatformStand=true; tHum.Sit=true
                if tick()-lastRemote>=REMOTE_DELAY then
                    lastRemote=tick()
                    pcall(function()
                        GE.SetNetworkOwner:FireServer(tRoot,lockPos)
                        GE.DestroyGrabLine:FireServer(tRoot)
                        local weld=R_Det:FindFirstChild("RightWeld") or R_Det:FindFirstChildWhichIsA("Weld")
                        if weld then CD:FireServer(weld); CG:FireServer(R_Det,tRoot,weld) end
                    end)
                end
            end
        else dragging=false; grabStartTime=0 end
        RunService.Heartbeat:Wait()
    end
    if blobRoot then blobRoot.CFrame=savedPos; blobRoot.Velocity=Vector3.zero end
    if CFG.loopKickBlobMode=="TP to Sky" then removeSkyKickFloor() end
end

local function setupShurikenESP()
    if not CFG.shurikenESPEnabled then return end
    local function addESPToPart(part)
        if part:FindFirstChild("RainbowESP") then return nil end
        local box=Instance.new("BoxHandleAdornment")
        box.Name="RainbowESP"; box.Adornee=part; box.Size=part.Size
        box.AlwaysOnTop=true; box.ZIndex=10; box.Transparency=0.3
        box.Color3=Color3.new(1,0,0); box.Parent=part; return box
    end
    local function createESPForShuriken(shuriken)
        if State.espObjects[shuriken] then return end
        local boxes={}
        for _, child in ipairs(shuriken:GetDescendants()) do
            if child:IsA("BasePart") then local box=addESPToPart(child); if box then table.insert(boxes,box) end end
        end
        if #boxes>0 then State.espObjects[shuriken]=boxes end
    end
    local function removeESPFromShuriken(shuriken)
        if not State.espObjects[shuriken] then return end
        for _, box in ipairs(State.espObjects[shuriken]) do if box then box:Destroy() end end
        State.espObjects[shuriken]=nil
    end
    local function findAndESPShurikens()
        local found={}
        for _, player in ipairs(Players:GetPlayers()) do
            local folder=Workspace:FindFirstChild(player.Name.."SpawnedInToys")
            if folder then
                for _, item in ipairs(folder:GetChildren()) do
                    if item:IsA("Model") and (item.Name=="CreatureBlobman" or item.Name=="AntiKick") then
                        found[item]=true; createESPForShuriken(item)
                    end
                end
            end
        end
        local pi=Workspace:FindFirstChild("PlotItems")
        if pi then
            for _, plot in ipairs(pi:GetChildren()) do
                for _, item in ipairs(plot:GetDescendants()) do
                    if item:IsA("Model") and (item.Name=="CreatureBlobman" or item.Name=="AntiKick") then
                        found[item]=true; createESPForShuriken(item)
                    end
                end
            end
        end
        for shuriken, _ in pairs(State.espObjects) do
            if not found[shuriken] or not shuriken.Parent then removeESPFromShuriken(shuriken) end
        end
    end
    local conn1=RunService.Heartbeat:Connect(function() if CFG.shurikenESPEnabled then findAndESPShurikens() end end)
    table.insert(State.visualConnections,conn1)
    local rainbowHue=0
    local conn2=RunService.RenderStepped:Connect(function(dt)
        if not CFG.shurikenESPEnabled then return end
        rainbowHue=(rainbowHue+dt*60*CFG.rainbowSpeedESP)%360
        local c=Color3.fromHSV(rainbowHue/360,1,1)
        for _, boxes in pairs(State.espObjects) do
            for _, box in ipairs(boxes) do if box and box.Parent then box.Color3=c end end
        end
    end)
    table.insert(State.visualConnections,conn2)
    return function()
        for shuriken, _ in pairs(State.espObjects) do removeESPFromShuriken(shuriken) end
        State.espObjects={}
    end
end

local function getTentacleMainPart(model)

    _G.TentacleActive=false
    for _, conn in ipairs(tentacleConnections) do pcall(function() conn:Disconnect() end) end
    tentacleConnections={}; tentacles={}

local function getTentSegmentPosition(rootCF,index,sx,sy,sz,sideOffset)
    local so=sideOffset or 0
    local baseOffset=CFrame.new(so,1,3)
    local bendFactor=index/TENT_SEGMENT_COUNT
    local segCF=rootCF*baseOffset
        *CFrame.Angles(math.rad(sy*tentBendStrength*bendFactor),math.rad(sx*tentBendStrength*bendFactor+tentTwistAngle*bendFactor),math.rad(sz*tentBendStrength*bendFactor))
        *CFrame.new(0,TENT_SEGMENT_SPACING*index,0)
    return segCF.Position
end

local function getTentSegmentCFrame(rootCF,index,sx,sy,sz,sideOffset)
    local thisPos=getTentSegmentPosition(rootCF,index,sx,sy,sz,sideOffset)
    local dir
    if index<TENT_SEGMENT_COUNT then
        dir=getTentSegmentPosition(rootCF,index+1,sx,sy,sz,sideOffset)-thisPos
    else
        dir=thisPos-getTentSegmentPosition(rootCF,index-1,sx,sy,sz,sideOffset)
    end
    if dir.Magnitude<0.001 then return CFrame.new(thisPos) end
    dir=dir.Unit
    local worldUp=Vector3.new(0,1,0)
    local right=worldUp:Cross(dir)
    right=(right.Magnitude<0.001) and Vector3.new(1,0,0) or right.Unit
    return CFrame.fromMatrix(thisPos,right,dir,-dir:Cross(right).Unit)
end

local function calcStickToTarget(rootCF,targetPos,sideOffset)
    local so=sideOffset or 0
    local basePos=(rootCF*CFrame.new(so,1,3)).Position
    local localDir=rootCF:VectorToObjectSpace((targetPos-basePos)).Unit
    return math.clamp(localDir.X*3,-3,3),math.clamp(localDir.Y*3,-3,3),math.clamp(-localDir.Z*3,-3,3)
end

    end
    task.wait(0.3)
    local totalSegments=TENT_SEGMENT_COUNT*StickCount

    for _=1,totalSegments do
        pcall(function() SpawnToyRemoteFunction:InvokeServer("TetracubeI",spawnBaseCF,Vector3.zero) end)

    end
    task.wait(0.3)
    local folder

    else folder=myFolder end
    if not folder then return end

    local allItems={}
    for _, item in ipairs(folder:GetChildren()) do
        if item.Name=="TetracubeI" and #allItems<totalSegments then
            local p=getTentacleMainPart(item)
            if p then
                pcall(function() for _, v in ipairs(item:GetDescendants()) do if v:IsA("BasePart") then v.CanCollide=false; v.CanTouch=false end end end)
                table.insert(allItems,{part=p,model=item})
            end
        end
    end
    local offsets=STICK_SIDE_OFFSETS[StickCount] or {0}
    for si=1,StickCount do
        local sideOffset=offsets[si] or 0; local tentParts={}
        for seg=1,TENT_SEGMENT_COUNT do
            local itemIndex=(si-1)*TENT_SEGMENT_COUNT+seg; local data=allItems[itemIndex]
            if data then
                local targetCF=getTentSegmentCFrame(hrp.CFrame,seg,Sticks[si].x,Sticks[si].y,Sticks[si].z,sideOffset)

                table.insert(tentParts,{part=data.part,model=data.model,index=seg,bp=bpT,bg=bgT})
            end
        end
        tentacles[si]={parts=tentParts,sideOffset=sideOffset}
    end
    local conn=RunService.Heartbeat:Connect(function(dt)
        if not _G.TentacleActive then return end
        local c=LocalPlayer.Character; local rRoot=c and c:FindFirstChild("HumanoidRootPart"); if not rRoot then return end
        local elapsed=tick()-startTime; local useNetOwner=(not isPlot) or (elapsed<5)
        if tentWiggleEnabled then tentWiggleTime=tentWiggleTime+dt*tentWiggleSpeed end
        for si=1,StickCount do

            local s=Sticks[si]; local so=tData.sideOffset; local sx,sy,sz=s.x,s.y,s.z

            end

                    if useNetOwner then pcall(function() SetNetworkOwner:FireServer(data.part,targetCF) end) end
                    data.bp.Position=targetCF.Position; data.bg.CFrame=targetCF

            end
        end

        task.wait()
        if CFG.characterWalkSpeedEnabled then
            pcall(function()
                local char=getChar(); local hrp=char and char:FindFirstChild("HumanoidRootPart"); local hum=char and char:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.MoveDirection.Magnitude>0 then
                    local cam=workspace.CurrentCamera.CFrame; local md=hum.MoveDirection
                    local cmv=cam:VectorToWorldSpace(Vector3.new(md.X,0,md.Z)); local mv=Vector3.new(cmv.X,0,cmv.Z).Unit
                    hrp.CFrame=hrp.CFrame+(mv*CFG.characterWalkSpeed)
                end
            end)
        end
    end
end)

-- RenderStepped
_G.spinAngle=0
RunService.RenderStepped:Connect(function(dt)
    if CFG.WingsActive and #State.Wings>0 then
        pcall(function()
            local Char=LocalPlayer.Character
            local torso=Char and (Char:FindFirstChild("Torso") or Char:FindFirstChild("UpperTorso"))
            if torso and Char:FindFirstChild("HumanoidRootPart") then
                CFG.TimeCounter=CFG.TimeCounter+dt*(CFG.WingSpeed+Char.HumanoidRootPart.Velocity.Magnitude/40)
                for i, Wing in ipairs(State.Wings) do
                    local dir=(i==1) and 1 or -1
                    local flap=math.sin(CFG.TimeCounter)*math.rad(CFG.WingAngle+(Char.HumanoidRootPart.Velocity.Magnitude/4))*dir
                    Wing.Handle.CFrame=torso.CFrame*CONST.Offsets[i]*CFrame.Angles(0,0,flap)
                    if Wing.Reserved then Wing.Reserved.BP.Position=Wing.Handle.Position; Wing.Reserved.BG.CFrame=Wing.Handle.CFrame*CFrame.Angles(math.rad(90),0,math.rad(90)) end
                    for idx, Seg in ipairs(Wing.Segments) do
                        local toFollow=(idx==1) and Wing.Handle.CFrame or Wing.Segments[idx-1].Part.CFrame
                        Seg.Part.CFrame=Seg.Part.CFrame:Lerp(toFollow*CONST.Offsets[i],0.5)
                        if Wing.Sync[idx] then Wing.Sync[idx].BP.Position=Seg.Part.Position; Wing.Sync[idx].BG.CFrame=Seg.Part.CFrame*CFrame.Angles(math.rad(90),0,math.rad(90)) end
                    end
                end
            end
        end)
    end
    if CFG.headEnabled and not CFG.danceEnabled then
        pcall(function()
            local char=getChar(); local hrp=GetRepresentativePart(char,"HumanoidRootPart"); local head=GetRepresentativePart(char,"Head")
            if hrp and head then
                _G.spinAngle=_G.spinAngle+CFG.speed*dt
                local c=hrp.Position
                head.CFrame=CFrame.new(Vector3.new(c.X+math.cos(_G.spinAngle)*CFG.radius,c.Y+CFG.height,c.Z+math.sin(_G.spinAngle)*CFG.radius))*(head.CFrame-head.CFrame.p)
            end
        end)
    end
    if CFG.characterSpinEnabled then
        pcall(function()
            local char=getChar(); local hrp=char and char:FindFirstChild("HumanoidRootPart")
            if hrp then hrp.CFrame=hrp.CFrame*CFrame.Angles(0,math.rad(CFG.characterSpinSpeed*100*dt),0) end
        end)
    end
end)

end)

-- Obey Logic Loop
task.spawn(function()
    while true do
        task.wait()
        if (CFG.obeyPartEnabled or CFG.obeyAllEnabled) and CFG.selectedPlayerName and not CFG.danceEnabled then
            pcall(function()
                local target=Players:FindFirstChild(CFG.selectedPlayerName); if not (target and target.Character) then return end
                local char=getChar()
                if CFG.obeyPartEnabled then
                    local tPart=GetRepresentativePart(target.Character,CFG.selectedPart); local myPart=GetRepresentativePart(char,CFG.selectedPart)
                    if tPart and myPart then myPart.CFrame=myPart.CFrame:Lerp(tPart.CFrame,CFG.lerpAlpha) end
                end
                if CFG.obeyAllEnabled then
                    local tHRP=GetRepresentativePart(target.Character,"HumanoidRootPart"); if not tHRP then return end
                    local offset=tHRP.CFrame.RightVector*CFG.obeyRightDistance
                    for _, key in ipairs({"Head","Left Arm","Right Arm","Left Leg","Right Leg"}) do
                        local tP=GetRepresentativePart(target.Character,key); local myP=GetRepresentativePart(char,key)
                        if tP and myP then myP.CFrame=myP.CFrame:Lerp(CFrame.new(tP.Position+offset)*(myP.CFrame-myP.CFrame.p),CFG.lerpAlpha) end
                    end
                end
            end)
        end
    end
end)

-- Feet TP+Delete Loop
task.spawn(function()
    while true do
        task.wait()
        if CFG.feetTPDeleteEnabled then
            pcall(function()
                local char=getChar(); local savedPos={}
                for _, p in pairs({"Head","Left Arm","Right Arm","HumanoidRootPart"}) do
                    local part=GetRepresentativePart(char,p); if part then savedPos[p]=part.CFrame end
                end
                local left=GetRepresentativePart(char,"Left Leg"); local right=GetRepresentativePart(char,"Right Leg")
                if left then left.CFrame=CFrame.new(0,999999999,0) end
                if right then right.CFrame=CFrame.new(0,999999999,0) end
                task.wait(0.1)
                if left then left:Destroy() end; if right then right:Destroy() end
                task.wait(0.1)
                for k, v in pairs(savedPos) do local part=GetRepresentativePart(char,k); if part then part.CFrame=v end end
                CFG.feetTPDeleteEnabled=false
            end)
        end
    end
end)
